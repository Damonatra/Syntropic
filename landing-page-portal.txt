function smooth(pts) {
  if (pts.length < 2) return '';
  let d = `M ${pts[0].x.toFixed(2)} ${pts[0].y.toFixed(2)}`;
  for (let i = 0; i < pts.length - 1; i++) {
    const p0 = pts[Math.max(0, i-1)], p1 = pts[i], p2 = pts[i+1], p3 = pts[Math.min(pts.length-1, i+2)];
    d += ` C ${(p1.x+(p2.x-p0.x)/6).toFixed(2)} ${(p1.y+(p2.y-p0.y)/6).toFixed(2)}, ${(p2.x-(p3.x-p1.x)/6).toFixed(2)} ${(p2.y-(p3.y-p1.y)/6).toFixed(2)}, ${p2.x.toFixed(2)} ${p2.y.toFixed(2)}`;
  }
  return d;
}
const SPARK_W = 96, SPARK_H = 28, N = 14;
class Spark {
  constructor(path) {
    this.p = path; this.dir = Number(path.dataset.dir || 1);
    this.phase = Number(path.dataset.phase || 0); this.freq = Number(path.dataset.freq || 4.2);
    this.cur = Array.from({ length: N }, (_, i) => this.pt(i / (N-1), Number(path.dataset.norm || .5)));
    this.raf = null; this.render();
  }
  hAt(x, v) {
    const base = this.dir > 0 ? lerp(0.30 + 0.30*v, 0.25 + 0.65*v, x) : lerp(0.80 - 0.15*v, 0.30 + 0.30*v, x);
    const wave = 0.14 * Math.sin(x * this.freq + this.phase + v * 2.6) * (0.35 + 0.65*x);
    return clamp(base + wave, 0.06, 0.96);
  }
  pt(x, v) { const h = this.hAt(x, v); return { x: 3 + x * (SPARK_W - 6), y: SPARK_H - 4 - h * (SPARK_H - 9) }; }
  pts(a) { return a.map((h, i) => ({ x: 3 + (i/(N-1))*(SPARK_W-6), y: SPARK_H - 4 - h*(SPARK_H-9) })); }
  render() { this.p.setAttribute('d', smooth(this.pts(this.cur))); }
  setTarget(v, dur = 420) {
    const target = Array.from({ length: N }, (_, i) => this.hAt(i/(N-1), v));
    if (RM) { this.cur = target; this.render(); return; }
    const from = this.cur.slice(), t0 = performance.now();
    if (this.raf) cancelAnimationFrame(this.raf);
    const step = t => { const p = clamp((t - t0)/dur, 0, 1), e = easeOut(p);
      this.cur = target.map((h, i) => lerp(from[i], h, e)); this.render();
      this.raf = p < 1 ? requestAnimationFrame(step) : null; };
    this.raf = requestAnimationFrame(step);
  }
  drawIn() {
    const L = this.p.getTotalLength(); if (!L || RM) return;
    this.p.style.strokeDasharray = L; this.p.style.strokeDashoffset = L;
    this.p.getBoundingClientRect(); this.p.style.transition = 'stroke-dashoffset 1s ease';
    this.p.style.strokeDashoffset = '0';
    setTimeout(() => { this.p.style.strokeDasharray = 'none'; this.p.style.transition = ''; }, 1100);
  }
}

const FOCUSABLE = 'a[href], button:not([disabled]), input:not([type="hidden"]):not([disabled]), select:not([disabled]), textarea:not([disabled]), [tabindex]:not([tabindex="-1"])';
function trapTab(e, root) {
  const items = $$(FOCUSABLE, root).filter(el => el.getClientRects().length > 0);
  if (!items.length) return;
  const first = items[0], last = items[items.length - 1];
  if (e.shiftKey && document.activeElement === first) { e.preventDefault(); last.focus(); }
  else if (!e.shiftKey && document.activeElement === last) { e.preventDefault(); first.focus(); }
}
document.addEventListener('keydown', e => {
  if (e.key === 'Escape') {
    if (!portalRoot.hidden) { closePortal(); return; }
    if (!modal.hidden) { closeModal(); return; }
  }
  if (e.key === 'Tab') {
    if (!portalRoot.hidden) trapTab(e, portalRoot);
    else if (!modal.hidden) trapTab(e, $('.modal-card', modal));
  }
});

function buildBusinessDaySlots() {
  const slots = []; const today = new Date(); let d = 0;
  while (slots.length < 3) {
    d++;
    const dt = new Date(today); dt.setDate(today.getDate() + d);
    const dow = dt.getDay(); if (dow === 0 || dow === 6) continue;

  }
  return slots;
}

function addFiles(list) {
  qWarn.hidden = true; let skipped = 0, added = 0;
  for (const f of list) {

  persist(); renderQueue();

const stages = $$('#pipeline .pipe-stage');
const USE_REAL_CLOCK = !!(CONFIG.checkoutEndpoint || CONFIG.ingestEndpoint);
const STAGE_REAL = [2 * 3600e3, 24 * 3600e3, 47.5 * 3600e3];
function computeStage(elapsed) {
  const marks = USE_REAL_CLOCK ? STAGE_REAL : [1, 2, 3].map(i => i * CONFIG.demoStageSeconds * 1000);
  let c = 1; for (const m of marks) if (elapsed >= m) c++;
  return Math.min(c, 4);
}
function setStage(done) {
  stages.forEach((el, i) => {
    el.classList.toggle('is-done', done === 4 || i < done);
    el.classList.toggle('is-active', i === done && done < 4);
  });
}

function stopAuditClock() { if (auditTimer) { clearInterval(auditTimer); auditTimer = null; } }

  const elapsed = Date.now() - pState.auditStartedAt;
