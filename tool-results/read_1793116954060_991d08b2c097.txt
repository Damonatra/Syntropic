   165→export function TeamCard({
   166→  initial, tone, name, role, isOnline, target, completed, discovery, auditsClosed, contracts,
   167→  isCurrentUser, onTargetChange,
   168→}: {
   169→  initial: string; tone: 'ink' | 'blush' | 'steel'; name: string; role: string; isOnline: boolean;
   170→  target: number; completed: number; discovery: number; auditsClosed: number; contracts: number;
   171→  isCurrentUser: boolean; onTargetChange?: (next: number) => void;
   172→}) {
   173→  const avatar = tone === 'ink' ? 'bg-ink text-paper' : tone === 'blush' ? 'bg-blush text-sienna' : 'bg-ink/85 text-paper';
   174→  const pct = target > 0 ? Math.min(100, Math.round((completed / target) * 100)) : 0;
   175→  return (
   176→    <div className="rounded-2xl bg-fog p-5">
   177→      <div className="flex items-start justify-between gap-3">
   178→        <div className="flex items-center gap-3">
   179→          <div className={`relative grid h-10 w-10 place-items-center rounded-full text-[15px] font-medium ${avatar}`}>
   180→            {initial}
   181→            <span
   182→              title={isOnline ? 'Online now' : 'Offline'}
   183→              className={`absolute -right-0.5 -top-0.5 h-2.5 w-2.5 rounded-full ring-2 ring-fog ${isOnline ? 'bg-sienna' : 'bg-smoke/50'}`}
   184→            />
   185→          </div>
   186→          <div>
   187→            <div className="text-[15px] font-medium text-ink">{name}</div>
   188→            <div className="text-[12px] text-smoke">{role}</div>
   189→          </div>
   190→        </div>
   191→        {isOnline ? <LiveDot /> : <span className="text-[11px] text-smoke">offline</span>}
   192→      </div>
   193→
   194→      {/* Quarterly outbound target */}
   195→      <div className="mt-4 rounded-xl bg-paper px-3.5 py-3">
   196→        <div className="flex items-center justify-between text-[12.5px]">
   197→          <span className="text-steel">Quarterly outbound</span>
   198→          <span className="font-medium tabular-nums text-ink">
   199→            {completed.toLocaleString()} / {target.toLocaleString()}
   200→          </span>
   201→        </div>
   202→        <div className="mt-2 h-1.5 w-full overflow-hidden rounded-full bg-mist">
   203→          <div className="h-full rounded-full bg-sienna transition-all duration-500" style={{ width: `${pct}%` }} />
   204→        </div>
   205→        <div className="mt-1.5 flex items-center justify-between text-[11px] text-smoke">
   206→          <span>Q4 progress</span>
   207→          <span className="tabular-nums">{pct}% completed</span>
   208→        </div>
   209→        {isCurrentUser && onTargetChange && (
   210→          <div className="mt-2.5 flex items-center gap-2 border-t border-hairline pt-2.5">
   211→            <label htmlFor={`target-${name}`} className="text-[11.5px] text-steel">Adjust my target:</label>
   212→            <input
   213→              id={`target-${name}`}
   214→              type="number"
   215→              min={1}
   216→              defaultValue={target}
   217→              onBlur={(e) => {
   218→                const next = Number(e.target.value);
   219→                if (Number.isFinite(next) && next >= 1 && next !== target) onTargetChange(next);
   220→                else e.target.value = String(target);
   221→              }}
   222→              onKeyDown={(e) => {
   223→                if (e.key === 'Enter') (e.target as HTMLInputElement).blur();
   224→              }}
   225→              className="h-7 w-20 rounded-lg border border-hairline bg-paper px-2 text-[12px] tabular-nums text-ink outline-none transition-shadow focus-visible:shadow-[0_0_0_2px_rgba(93,42,26,0.25)]"
   226→            />
   227→          </div>
   228→        )}
   229→      </div>
   230→
   231→      <div className="mt-3 space-y-2">
   232→        {([
   233→          ['Discovery calls booked', discovery.toLocaleString()],
   234→          ['$3,500 audits sold', auditsClosed.toLocaleString()],
   235→          ['$50k contracts converted', contracts.toLocaleString()],
   236→        ] as [string, string][]).map(([k, v]) => (
   237→          <div key={k} className="flex items-center justify-between gap-3 rounded-xl bg-paper px-3.5 py-2.5 text-[13px]">
   238→            <span className="text-steel">{k}</span>
   239→            <span className="font-medium tabular-nums text-ink">{v}</span>
   240→          </div>
   241→        ))}
   242→      </div>
   243→    </div>
   244→  );
   245→}
