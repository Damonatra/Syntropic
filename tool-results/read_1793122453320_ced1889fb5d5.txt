    45→const relTime = (iso: string) => {
    46→  const mins = Math.max(1, Math.round((Date.now() - new Date(iso).getTime()) / 60000));
    47→  if (mins < 60) return `${mins} min${mins === 1 ? '' : 's'} ago`;
    48→  return `${Math.round(mins / 60)}h ago`;
