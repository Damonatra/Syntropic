   405→/* ── Show-the-work receipts (Feature: every number shows its work) ──
   406→ * Unfolds the exact ledger rows behind a figure, fetched from
   407→ * /api/ledger/receipts. Collapsed by default; fetched on first open. */
   408→const receiptKindLabel = (kind: string) =>
   409→  kind === 'MARGIN_DEFENDED' ? 'Margin defended' : kind === 'SETUP_FEE' ? 'Setup fee' : kind === 'AUDIT_FEE' ? 'Audit fee' : kind;
   410→
   411→export function ReceiptsDisclosure({
   412→  tenant, metric = 'defended', label = 'Show the work',
   413→}: {
   414→  tenant: string | null; metric?: 'defended' | 'revshare'; label?: string;
   415→}) {
   416→  const [open, setOpen] = useState(false);
   417→  const [data, setData] = useState<ReceiptsView | null>(null);
   418→  const [error, setError] = useState(false);
   419→
   420→  const toggle = async () => {
   421→    const next = !open;
   422→    setOpen(next);
   423→    if (next && !data && !error) {
   424→      try {
   425→        const params = new URLSearchParams();
   426→        if (tenant) params.set('tenant', tenant);
   427→        params.set('metric', metric);
   428→        const res = await fetch(`/api/ledger/receipts?${params.toString()}`, { cache: 'no-store' });
   429→        if (!res.ok) throw new Error(String(res.status));
   430→        setData((await res.json()) as ReceiptsView);
   431→      } catch {
   432→        setError(true);
   433→      }
   434→    }
   435→  };
   436→
   437→  return (
   438→    <div className="border-t border-hairline pt-2.5">
   439→      <button
   440→        type="button"
   441→        onClick={toggle}
   442→        aria-expanded={open}
   443→        className="inline-flex items-center gap-1.5 text-[12px] text-smoke transition-colors hover:text-sienna"
   444→      >
   445→        <ReceiptText className="h-3.5 w-3.5" strokeWidth={1.75} />
   446→        {label}
   447→        <ChevronDown className={`h-3 w-3 transition-transform duration-200 ${open ? 'rotate-180' : ''}`} strokeWidth={2} />
   448→      </button>
   449→
