     3→/**
     4→ * Layer 3 — Executive Enclave portal (client-facing twin of the Command Tower).
     5→ * Auth model: exactly one enclave per session. The tenant is fixed by the
     6→ * server (session cookie scoped to /portal); there is no switcher. The
     7→ * direct line streams over /portal/api/messages — session-scoped, so a
     8→ * client can only ever read/write its own thread — while the Backbone's
     9→ * unified inbox (R-06) sees client messages as unread within seconds.
    10→ * All financials arrive server-computed from /portal/api/enclave (R-01).
    11→ *
    12→ * Task 19 — trust as the product: the enclave now carries the reconciliation
    13→ * stamp (the same self-audit the tower runs), the client's money trail
    14→ * (recentLedger rendered, no longer a dead payload), expandable per-intercept
    15→ * receipts, genuinely-upcoming calls with one-click Join, a document shelf
    16→ * with upload + download, and Seen receipts on the direct line. Live updates
    17→ * ride the session-scoped SSE bus (/portal/api/bus) with an honest polling
    18→ * fallback; the feed itself reconciles every 30s.
    19→ */

    46→  new Date(iso).toLocaleTimeString('en-US', { hour: 'numeric', minute: '2-digit' });
    47→

    49→  new Date(iso).toLocaleDateString('en-US', { weekday: 'short', month: 'short', day: 'numeric' });

    51→const isToday = (iso: string) =>
    52→  new Date(iso).toDateString() === new Date().toDateString();

    72→export default function EnclavePortal({
    73→  tenantName,
    74→  contactName,
    75→}: {
    76→  tenantName: string;
    77→  contactName: string;
    78→}) {
    79→  const router = useRouter();
    80→  const [enclave, setEnclave] = useState<PortalEnclaveView | null>(null);
    81→  const [messages, setMessages] = useState<MessageView[]>([]);
    82→  const [draft, setDraft] = useState('');
    83→  const [sending, setSending] = useState(false);
    84→  const [loadError, setLoadError] = useState(false);
    85→  const [streamLive, setStreamLive] = useState(false);
    86→  const [expandedId, setExpandedId] = useState<string | null>(null);
    87→  const [copiedCallId, setCopiedCallId] = useState<string | null>(null);
    88→  const [uploading, setUploading] = useState(false);
    89→  const [shelfNote, setShelfNote] = useState<{ ok: boolean; text: string } | null>(null);
    90→  const [downloadingId, setDownloadingId] = useState<string | null>(null);
    91→  const fileRef = useRef<HTMLInputElement>(null);
    92→  const scrollRef = useRef<HTMLDivElement>(null);
    93→
    94→  const fetchJson = useCallback(async <T,>(url: string, init?: RequestInit): Promise<T> => {
    95→    const res = await fetch(url, { cache: 'no-store', ...init });
    96→    if (res.status === 401) {
    97→      // Session expired — hard redirect to sign-in.
    98→      window.location.href = '/portal/login';
    99→      throw new Error('unauthenticated');
   100→    }
   101→    if (!res.ok) throw new Error(`${url} → ${res.status}`);
   102→    return res.json() as Promise<T>;
   103→  }, []);
   104→
   105→  const refreshEnclave = useCallback(async () => {
   106→    try {
   107→      setEnclave(await fetchJson<PortalEnclaveView>('/portal/api/enclave'));
   108→      setLoadError(false);
   109→    } catch {
   110→      setLoadError(true);
   111→    }
   112→  }, [fetchJson]);
   113→
   114→  const refreshMessages = useCallback(async () => {
   115→    try {
   116→      setMessages(await fetchJson<MessageView[]>('/portal/api/messages'));
   117→    } catch {
   118→      /* transient — next tick retries (401 already redirects) */
   119→    }
   120→  }, [fetchJson]);
   121→
   122→  // Enclave data: on mount + 30s reconciliation poll (SSE covers the pushes).
   123→  useEffect(() => {
   124→    refreshEnclave().catch(() => undefined);
   125→    const slow = setInterval(() => refreshEnclave().catch(() => undefined), 30_000);
   126→    return () => clearInterval(slow);
   127→  }, [refreshEnclave]);
   128→
   129→  // Direct line: initial load, then the session-scoped SSE bus pushes.
   130→  useEffect(() => {
   131→    refreshMessages().catch(() => undefined);
   132→  }, [refreshMessages]);
   133→
   134→  useEffect(() => {
   135→    const es = new EventSource('/portal/api/bus');
   136→    es.onopen = () => setStreamLive(true);
   137→    es.onerror = () => setStreamLive(false); // EventSource retries; polling takes over below
   138→    es.addEventListener('message', () => void refreshMessages());
   139→    es.addEventListener('thread-read', () => void refreshMessages());
   140→    es.addEventListener('defense', () => void refreshEnclave());
   141→    es.addEventListener('hitl', () => void refreshEnclave());
   142→    return () => es.close();
   143→  }, [refreshMessages, refreshEnclave]);
   144→
   145→  // Honest fallback — while the stream is down, the thread polls every 5s.
   146→  useEffect(() => {
   147→    if (streamLive) return;
   148→    const t = setInterval(() => refreshMessages().catch(() => undefined), 5_000);
   149→    return () => clearInterval(t);
   150→  }, [streamLive, refreshMessages]);
   151→
   152→  // Keep the thread pinned to the latest message.
   153→  useEffect(() => {
   154→    scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight });
   155→  }, [messages.length]);
   156→
   157→  const send = async () => {
   158→    const text = draft.trim();
   159→    if (!text || sending) return;
   160→    setSending(true);
   161→    setDraft('');
   162→    try {
   163→      // Identity comes from the session server-side — only text travels.
   164→      await fetchJson('/portal/api/messages', {
   165→        method: 'POST',
   166→        headers: { 'Content-Type': 'application/json' },
   167→        body: JSON.stringify({ text }),
   168→      });
   169→      await refreshMessages();
   170→    } catch {
   171→      /* next poll reconciles */
   172→    } finally {
   173→      setSending(false);
   174→    }
   175→  };
   176→
   177→  const onUpload = async (files: FileList | null) => {
   178→    if (!files || files.length === 0) return;
   179→    setUploading(true);
   180→    setShelfNote(null);
   181→    try {
   182→      const form = new FormData();
   183→      Array.from(files).forEach((f) => form.append('files', f));
   184→      const res = await fetch('/portal/api/ingest', { method: 'POST', body: form });
   185→      const data = (await res.json().catch(() => null)) as
   186→        | { filesQueued?: number; hitlCount?: number; perFileSummary?: { lines: number }[]; detail?: string; error?: string }
   187→        | null;
   188→      if (res.status === 402) {
   189→        setShelfNote({ ok: false, text: data?.detail ?? 'File ingest requires an active deposit — ask your operator.' });
   190→      } else if (res.status === 429) {
   191→        setShelfNote({ ok: false, text: 'Too many uploads this hour — try again a little later.' });
   192→      } else if (!res.ok) {
   193→        setShelfNote({ ok: false, text: data?.detail ?? data?.error ?? 'Upload did not land — try again.' });
   194→      } else {
   195→        const lines = (data?.perFileSummary ?? []).reduce((a, x) => a + x.lines, 0);
   196→        const hitl = data?.hitlCount ?? 0;
   197→        setShelfNote({
   198→          ok: true,
   199→          text: `${data?.filesQueued ?? 0} file${data?.filesQueued === 1 ? '' : 's'} on the shelf — ${lines} lines parsed${hitl ? `, ${hitl} flagged for your team to review` : ''}.`,
   200→        });
   201→        await refreshEnclave();
   202→      }
   203→    } catch {
   204→      setShelfNote({ ok: false, text: 'Network error — the upload did not land.' });
   205→    } finally {
   206→      setUploading(false);
   207→      if (fileRef.current) fileRef.current.value = '';
   208→    }
   209→  };
   210→
   211→  const download = async (d: TenantDocumentView) => {
   212→    setDownloadingId(d.id);
   213→    setShelfNote(null);
   214→    try {
   215→      const res = await fetch(`/portal/api/documents/${d.id}`, { cache: 'no-store' });
   216→      if (!res.ok) {
   217→        const data = (await res.json().catch(() => null)) as { detail?: string } | null;
   218→        setShelfNote({ ok: false, text: data?.detail ?? `Could not retrieve ${d.fileName}.` });
   219→        return;
   220→      }
   221→      const blob = await res.blob();
   222→      const url = URL.createObjectURL(blob);
   223→      const a = document.createElement('a');
   224→      a.href = url;
   225→      a.download = d.fileName;
   226→      a.click();
   227→      URL.revokeObjectURL(url);
   228→    } catch {
   229→      setShelfNote({ ok: false, text: `Network error — ${d.fileName} did not download.` });
   230→    } finally {
   231→      setDownloadingId(null);
   232→    }
   233→  };
   234→
   235→  const copyJoin = async (callId: string, url: string) => {
   236→    try {
   237→      await navigator.clipboard.writeText(url);
   238→      setCopiedCallId(callId);
   239→      setTimeout(() => setCopiedCallId((c) => (c === callId ? null : c)), 2200);
   240→    } catch {
   241→      /* clipboard unavailable — the Join link remains visible */
   242→    }
   243→  };
   244→
   245→  const signOut = async () => {
   246→    await fetch('/portal/api/logout', { method: 'POST' }).catch(() => undefined);
   247→    router.push('/portal/login');
   248→    router.refresh();
   249→  };
   250→
   251→  const tenant = enclave?.tenant;
   252→  const thread = messages;
   253→  const documents = enclave?.documents ?? [];
   254→  const recon = enclave?.reconciliation ?? null;
   255→
   256→  return (
   257→    <div className="flex min-h-screen flex-col bg-paper text-ink selection:bg-blush selection:text-sienna">
   258→      {/* ── Masthead ───────────────────────────────────────────────── */}
   259→      <header className="sticky top-0 z-50 border-b border-hairline bg-paper/85 backdrop-blur-md">
   260→        <div className="mx-auto flex h-16 w-full max-w-[1120px] items-center justify-between px-5 sm:px-8 lg:px-10">
   261→          <div className="flex min-w-0 items-center gap-4">
   262→            <Link href="/portal" className="shrink-0 font-display text-[21px] leading-none tracking-[-0.01em] text-ink">
   263→              enclave
   264→            </Link>
   265→            <span className="hidden text-[13px] text-smoke sm:inline">Layer 03 · client boundary</span>
   266→            <span className="hidden items-center gap-2 rounded-full border border-hairline px-3 py-1.5 text-[12px] text-steel md:inline-flex">
   267→              <LiveDot />
   268→              {streamLive ? 'Enclave live · streamed' : 'Enclave live · polling'}
   269→            </span>
   270→          </div>
   271→          <div className="flex shrink-0 items-center gap-3">
   272→            <span className="hidden text-[13px] text-steel sm:inline">
   273→              Signed in as <span className="font-medium text-ink">{contactName}</span>
   274→              {tenant ? ` · ${tenant.name}` : ''}
   275→            </span>
   276→            <button
   277→              onClick={signOut}
   278→              className="inline-flex items-center gap-1.5 rounded-full border border-hairline px-3 py-1.5 text-[12.5px] font-medium text-steel transition-colors hover:bg-mist hover:text-ink"
   279→            >
   280→              <LogOut className="h-3.5 w-3.5" strokeWidth={1.75} />
   281→              Sign out
   282→            </button>
   283→          </div>
   284→        </div>
   285→      </header>
   286→
   287→      {/* ── Body ───────────────────────────────────────────────────── */}
   288→      <main className="mx-auto w-full max-w-[1120px] flex-1 px-5 py-10 sm:px-8 lg:px-10">
   289→        <ViewHeader
   290→          tag={`Executive enclave · ${tenant?.erp ?? '—'}`}
   291→          title="Your margin,"
   292→          accent="defended."
   293→          rest=""
   294→          sub={
   295→            tenant
   296→              ? `${tenant.name} — ${tenant.status.toLowerCase()}. Every intercepted dollar below is computed from the same ledger your Syntropic team reports against.`
   297→              : 'Loading your enclave…'
   298→          }
   299→        />
   300→
   301→        {loadError && (
   302→          <div className="mb-6 rounded-2xl border border-hairline bg-mist p-4 text-[14px] text-steel">
   303→            Enclave feed unreachable — retrying automatically. If this persists, contact your Syntropic owner ({tenant?.owner ?? 'team'}).
   304→          </div>
   305→        )}
   306→
   307→        {/* ── Reconciliation strip ───────────────────────────────────── */}
   308→        <section className={`${ARTIFACT} mt-6 flex flex-wrap items-center gap-x-6 gap-y-3 px-6 py-4`} aria-label="Reconciliation stamp">
   309→          <div className="flex items-center gap-2.5">
   310→            {recon && !recon.ok ? (
   311→              <AlertTriangle className="h-4 w-4 shrink-0 text-sienna" strokeWidth={1.75} />
   312→            ) : (
   313→              <ShieldCheck className={`h-4 w-4 shrink-0 ${recon?.ok ? 'text-sienna' : 'text-smoke'}`} strokeWidth={1.75} />
   314→            )}
   315→            <div>
   316→              <div className="text-[14px] font-medium text-ink">Reconciled ledger</div>
   317→              <div className="text-[12px] text-steel">The same self-audit your team runs — printed on your numbers.</div>
   318→            </div>
   319→          </div>
   320→          <div className="ml-auto flex flex-wrap items-center gap-2">
   321→            {recon ? (
   322→              <>
   323→                {recon.checks != null && (
   324→                  <span className="inline-flex items-center rounded-full border border-hairline px-3 py-1 text-[12px] font-medium text-ink tabular-nums">
   325→                    {recon.checks} independent checks
   326→                  </span>
   327→                )}
   328→                {recon.drift != null && (
   329→                  <span
   330→                    className={`inline-flex items-center rounded-full px-3 py-1 text-[12px] font-medium tabular-nums ${
   331→                      recon.ok ? 'bg-blush text-sienna' : 'border border-hairline text-steel'
   332→                    }`}
   333→                  >
   334→                    {recon.drift} drift
   335→                  </span>
   336→                )}
   337→                <span className="inline-flex items-center gap-1.5 text-[12px] text-steel">
   338→                  <LiveDot />
   339→                  stamped {stampWhen(recon.stampedAt)}
   340→                </span>
   341→                {!recon.ok && <span className="w-full text-[12px] leading-relaxed text-sienna">{recon.detail}</span>}
   342→              </>
   343→            ) : (
   344→              <span className="text-[12.5px] text-smoke">
   345→                Self-audit has not stamped yet — every figure on this page still comes straight off the ledger.
   346→              </span>
   347→            )}
   348→          </div>
   349→        </section>
   350→
   351→        {/* ── Enclave stats ──────────────────────────────────────────── */}
   352→        <div className="mt-6 grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-4">
   353→          <div className={`${ARTIFACT} bg-blush p-5`}>
   354→            <div className="text-[13px] text-sienna/80">Margin recovered</div>
   355→            <div className="mt-2 font-display text-[30px] leading-none tracking-[-0.01em] text-sienna tabular-nums">
   356→              {enclave ? fmtUsd(enclave.marginDefendedCents) : '—'}
   357→            </div>
   358→            <div className="mt-2 text-[12px] text-sienna/70">
   359→              {enclave && enclave.last24hDeltaCents > 0 ? (
   360→                <span className="inline-flex items-center gap-1 font-medium tabular-nums">
   361→                  <TrendingUp className="h-3 w-3" strokeWidth={2} />+{fmtUsd(enclave.last24hDeltaCents, true)} in the last 24h
   362→                </span>
   363→              ) : (
   364→                'Lifetime, this enclave'
   365→              )}
   366→            </div>
   367→          </div>
   368→          <div className={`${ARTIFACT} p-5`}>
   369→            <div className="text-[13px] text-steel">Defenses logged</div>
   370→            <div className="mt-2 font-display text-[30px] leading-none tracking-[-0.01em] text-ink tabular-nums">
   371→              {enclave ? enclave.defensesLogged : '—'}
   372→            </div>
   373→            <div className="mt-2 text-[12px] text-smoke">Approved line-item defenses</div>
   374→          </div>
   375→          <div className={`${ARTIFACT} p-5`}>
   376→            <div className="text-[13px] text-steel">Guarantee status</div>
   377→            <div className="mt-2 text-[15px] font-medium leading-snug text-ink">{tenant?.guaranteeLabel ?? '—'}</div>
   378→            <div className="mt-2 inline-flex items-center gap-1.5 text-[12px] text-steel">
   379→              <ShieldCheck className={`h-3.5 w-3.5 ${tenant?.guaranteePassed ? 'text-sienna' : 'text-smoke'}`} strokeWidth={1.75} />
   380→              {tenant?.guaranteePassed ? 'Baseline passed' : 'Diagnostic pending'}
   381→            </div>
   382→          </div>
   383→          <div className={`${ARTIFACT} p-5`}>
   384→            <div className="text-[13px] text-steel">Deposit / fee</div>
   385→            <div className="mt-2 text-[15px] font-medium leading-snug text-ink">{tenant?.depositLabel ?? '—'}</div>
   386→            <div className="mt-2 text-[12px] text-smoke">{tenant?.stage ?? ''}</div>
   387→          </div>
   388→        </div>

   390→        {/* ── Feed + direct line ─────────────────────────────────────── */}
   391→        <div className="mt-6 grid grid-cols-1 gap-6 lg:grid-cols-12">
   392→          {/* Live intercept feed + money trail */}
   393→          <div className="space-y-6 lg:col-span-7">
   394→            <section className={`${ARTIFACT} p-6`} aria-label="Live enclave feed">

   397→                <span className="inline-flex items-center gap-2 text-[12px] text-steel">
