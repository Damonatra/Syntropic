   408→/* ── Show-the-work receipts (Feature: every number shows its work) ──
   409→ * Unfolds the exact ledger rows behind a figure, fetched from
   410→ * /api/ledger/receipts. Collapsed by default; fetched on first open. */
   411→const receiptKindLabel = (kind: string) =>
   412→  kind === 'MARGIN_DEFENDED' ? 'Margin defended' : kind === 'SETUP_FEE' ? 'Setup fee' : kind === 'AUDIT_FEE' ? 'Audit fee' : kind;
   413→
   414→export function ReceiptsDisclosure({
   415→  tenant, metric = 'defended', label = 'Show the work',
   416→}: {
   417→  tenant: string | null; metric?: 'defended' | 'revshare'; label?: string;
   418→}) {
   419→  const [open, setOpen] = useState(false);
   420→  const [data, setData] = useState<ReceiptsView | null>(null);
   421→  const [error, setError] = useState(false);
   422→
   423→  const toggle = async () => {
   424→    const next = !open;
   425→    setOpen(next);
   426→    if (next && !data && !error) {
   427→      try {
   428→        const params = new URLSearchParams();
   429→        if (tenant) params.set('tenant', tenant);
   430→        params.set('metric', metric);
   431→        const res = await fetch(`/api/ledger/receipts?${params.toString()}`, { cache: 'no-store' });
   432→        if (!res.ok) throw new Error(String(res.status));
   433→        setData((await res.json()) as ReceiptsView);
   434→      } catch {
   435→        setError(true);
   436→      }
   437→    }
   438→  };
   439→
   440→  return (
   441→    <div className="border-t border-hairline pt-2.5">
   442→      <button
   443→        type="button"
   444→        onClick={toggle}
   445→        aria-expanded={open}
   446→        className="inline-flex items-center gap-1.5 text-[12px] text-smoke transition-colors hover:text-sienna"
   447→      >
