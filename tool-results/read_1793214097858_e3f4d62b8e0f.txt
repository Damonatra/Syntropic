     1→"""Module 6 — Automated Mock Audit Screenshot Generator.
     2→
     3→Captures a high-resolution headless screenshot of the prospect's public
     4→RFQ / contact / catalog page (Playwright, Chromium, 2× device scale), then
     5→composites a soft-tactile audit calculator widget on top of the image that
     6→displays the estimated annual net-margin leakage for their vertical.
     7→
     8→Visual standard
     9→---------------
    10→The widget is composed strictly from the achromatic Steep design tokens in
    11→``DESIGN.md`` — Paper White surface, hairline border, soft 10%-opacity
    12→shadow, 20px radius — honouring the standing "only shades of grey and
    13→black" doctrine. No chromatic color exists anywhere in the pipeline.
    14→
    15→Outputs
    16→-------
    17→* ``prospects.audit_image_url`` — public CDN link, embedded dynamically in
    18→  Channel A cold emails and the Josias high-touch task queue.
    19→* ``outbox`` event — ``audit.preview_generated``.
    20→
    21→Storage
    22→-------
    23→* ``s3``    — any S3-compatible bucket (AWS S3, Cloudflare R2, MinIO) served
    24→  behind ``audit_preview.cdn_base_url``.
    25→* ``local`` — a writable directory the portal/CDN mounts at
    26→  ``<portal_base_url>/assets``.
    27→
    28→Usage
    29→-----
    30→    python -m vertebra.intelligence.audit_preview --prospect-id <uuid>
    31→    python -m vertebra.intelligence.audit_preview --batch --limit 20
    32→
    33→Requires: ``playwright install chromium`` once per worker image.
    34→"""

    56→# --------------------------------------------------------------------------- #
    57→# Deterministic leakage model — vertical benchmark table (v1)
    58→# --------------------------------------------------------------------------- #
    59→#
    60→# The estimator is deliberately transparent: monthly RFQ volume is scaled
    61→# from the prospect's revenue estimate, then annual leakage = volume × 12 ×
    62→# average order value × gross margin × capture-loss rate. The benchmark
    63→# rows are conservative published-band values for distribution verticals and
    64→# the full formula travels with every estimate (`methodology`) so any figure
    65→# on an audit image can be defended line by line.
    66→
    67→LEAKAGE_BENCHMARKS: dict[str, dict[str, float]] = {
    68→    "pvf": {
    69→        "rfqs_per_month_per_10m": 90, "avg_order_value": 4_200, "gross_margin": 0.27,
    70→        "capture_loss": 0.18, "quote_delay_hours": 26,
    71→    },
    72→    "electrical": {
    73→        "rfqs_per_month_per_10m": 85, "avg_order_value": 1_800, "gross_margin": 0.22,
    74→        "capture_loss": 0.15, "quote_delay_hours": 18,
    75→    },
    76→    "hvac": {
    77→        "rfqs_per_month_per_10m": 70, "avg_order_value": 3_100, "gross_margin": 0.31,
    78→        "capture_loss": 0.16, "quote_delay_hours": 22,
    79→    },
    80→    "fluid_power": {
    81→        "rfqs_per_month_per_10m": 55, "avg_order_value": 5_200, "gross_margin": 0.33,
    82→        "capture_loss": 0.19, "quote_delay_hours": 30,
    83→    },
    84→}
    85→
    86→#: Conservative fallback for prospects without a known vertical.
    87→_DEFAULT_BENCHMARK: dict[str, float] = {
    88→    "rfqs_per_month_per_10m": 65, "avg_order_value": 2_600, "gross_margin": 0.24,
    89→    "capture_loss": 0.15, "quote_delay_hours": 24,
    90→}
    91→
    92→#: Midpoint of the standing $10M–$150M ICP corridor when revenue is unknown.
    93→_DEFAULT_REVENUE_BASIS: float = 60_000_000.0
    94→
    95→_VERTICAL_LABELS: dict[str, str] = {
    96→    "pvf": "PVF",
    97→    "electrical": "Electrical",
    98→    "hvac": "HVAC",
    99→    "fluid_power": "Fluid Power",
   100→}
   101→
   102→
   103→def estimate_revenue_basis(revenue_estimate_usd: int | None) -> float:
   104→    """Revenue the leakage figure is scaled against (ICP midpoint fallback)."""
   105→    return float(revenue_estimate_usd) if revenue_estimate_usd else _DEFAULT_REVENUE_BASIS
   106→
   107→
   108→def build_leakage_estimate(
   109→    *, vertical: str | None, revenue_estimate_usd: int | None
   110→) -> LeakageEstimate:
   111→    """Deterministic annual net-margin leakage estimate for one prospect."""
   112→    benchmark = LEAKAGE_BENCHMARKS.get((vertical or "").lower(), _DEFAULT_BENCHMARK)
   113→    revenue_basis = estimate_revenue_basis(revenue_estimate_usd)
   114→    monthly_rfq_volume = max(1, int(round(benchmark["rfqs_per_month_per_10m"] * revenue_basis / 10_000_000)))
   115→    annual_leakage = (
   116→        monthly_rfq_volume
   117→        * 12
   118→        * benchmark["avg_order_value"]
   119→        * benchmark["gross_margin"]
   120→        * benchmark["capture_loss"]
   121→    )
   122→    methodology = (
   123→        f"{monthly_rfq_volume} RFQs/mo × 12 × ${benchmark['avg_order_value']:,.0f} avg order × "
   124→        f"{benchmark['gross_margin']:.0%} gross margin × {benchmark['capture_loss']:.0%} capture loss "
   125→        f"on manually-handled quotes (benchmark table audit_preview/v1, vertical="
   126→        f"{(vertical or 'unknown').lower()}, revenue basis ${revenue_basis / 1e6:.1f}M)"
   127→    )
   128→    return LeakageEstimate(
   129→        annual_net_margin_leakage=round(annual_leakage, 2),
   130→        revenue_basis_usd=revenue_basis,
   131→        monthly_rfq_volume=monthly_rfq_volume,
   132→        avg_order_value_usd=benchmark["avg_order_value"],
   133→        gross_margin_pct=benchmark["gross_margin"],
   134→        capture_loss_rate=benchmark["capture_loss"],
   135→        avg_quote_delay_hours=benchmark["quote_delay_hours"],
   136→        methodology=methodology,
   137→    )

   140→# --------------------------------------------------------------------------- #
   141→# Playwright capture
   142→# --------------------------------------------------------------------------- #
   143→
   144→
   145→@dataclass(slots=True)
   146→class _CaptureResult:
   147→    png: bytes
   148→    width: int
   149→    height: int
   150→
   151→
   152→class ScreenshotCapture:
   153→    """Headless Chromium capture at high resolution (device-scale 2×)."""
   154→
   155→    def __init__(self, settings: AuditPreviewSettings | None = None) -> None:
   156→        self._settings = settings or get_settings().audit_preview
   157→
   158→    async def capture(self, url: str) -> _CaptureResult:
   159→        try:
   160→            from playwright.async_api import TimeoutError as PWTimeoutError
   161→            from playwright.async_api import async_playwright
   162→        except ImportError as exc:  # pragma: no cover - environment guard
   163→            raise AuditPreviewError(
   164→                "playwright is not installed — run 'playwright install chromium'"
   165→            ) from exc
   166→
   167→        viewport = {"width": self._settings.viewport_width, "height": self._settings.viewport_height}
   168→        dsf = self._settings.device_scale_factor
   169→        try:
   170→            async with async_playwright() as pw:
   171→                browser = await pw.chromium.launch(
   172→                    headless=True, args=["--no-sandbox", "--disable-dev-shm-usage"]
   173→                )
   174→                try:
   175→                    page = await browser.new_page(viewport=viewport, device_scale_factor=dsf)
   176→                    try:
   177→                        await page.goto(
   178→                            url, wait_until="networkidle", timeout=self._settings.navigation_timeout_ms
   179→                        )
   180→                    except PWTimeoutError:
   181→                        log.warning("capture_networkidle_timeout_falling_back", url=url)
   182→                        await page.goto(url, wait_until="load", timeout=20_000)
   183→                    # Nudge lazy-loaded catalog imagery, then settle at the top.
   184→                    await page.evaluate("window.scrollTo(0, Math.floor(document.body.scrollHeight / 4))")
   185→                    await page.wait_for_timeout(900)
   186→                    await page.evaluate("window.scrollTo(0, 0)")
   187→                    await page.wait_for_timeout(350)
   188→                    png = await page.screenshot(type="png")
   189→                    return _CaptureResult(
   190→                        png=png, width=viewport["width"] * dsf, height=viewport["height"] * dsf
   191→                    )
   192→                finally:
   193→                    await browser.close()
   194→        except AuditPreviewError:
   195→            raise
   196→        except Exception as exc:  # noqa: BLE001 - normalise playwright errors
   197→            raise AuditPreviewError(
   198→                f"capture failed for {url}: {type(exc).__name__}: {exc}", context={"url": url}
   199→            ) from exc

   202→# --------------------------------------------------------------------------- #
   203→# Widget overlay — Steep design tokens, achromatic only
   204→# --------------------------------------------------------------------------- #
   205→
   206→
   207→class _Tokens:
   208→    """Achromatic subset of the DESIGN.md (Steep) palette."""
   209→
   210→    INK = (23, 25, 28, 255)        # #17191c ink black — primary text
   211→    PAPER = (255, 255, 255, 255)   # #ffffff paper white — card surface
   212→    MIST = (242, 242, 243, 255)    # #f2f2f3 mist gray — gauge track
   213→    SLATE = (119, 123, 134, 255)   # #777b86 slate gray — captions
   214→    ASH = (151, 151, 153, 255)     # #979799 ash gray — eyebrow/footer
   215→    SMOKE = (163, 166, 175, 255)   # #a3a6af smoke gray — meta rows
   216→    HAIRLINE = (23, 25, 28, 22)    # ink at ~8% — artifact border
   217→    SHADOW = (10, 12, 15, 26)      # ink at ~10% — floating artifact shadow
   218→
   219→
   220→_FONT_CANDIDATES_BOLD = (
   221→    "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf",
   222→    "/usr/share/fonts/truetype/liberation/LiberationSans-Bold.ttf",
   223→)
   224→_FONT_CANDIDATES_REGULAR = (
   225→    "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
   226→    "/usr/share/fonts/truetype/liberation/LiberationSans-Regular.ttf",
   227→)
   228→
   229→
   230→class AuditWidgetOverlay:
   231→    """Composites the soft-tactile audit calculator card onto a screenshot."""
   232→
   233→    # Geometry is designed in CSS pixels and scaled by the device scale factor.
   234→    CARD_W = 480
   235→    CARD_H = 224
   236→    RADIUS = 20
   237→    MARGIN = 40
   238→
   239→    def __init__(self, settings: AuditPreviewSettings | None = None) -> None:
   240→        self._settings = settings or get_settings().audit_preview
   241→        self._font_cache: dict[tuple[str, int], Any] = {}
   242→
   243→    # ------------------------------------------------------------------ fonts
   244→    def _font(self, size_px: int, *, bold: bool = False) -> Any:
   245→        from PIL import ImageFont
   246→
   247→        candidates = _FONT_CANDIDATES_BOLD if bold else _FONT_CANDIDATES_REGULAR
   248→        key = (candidates[0], size_px)
   249→        if key not in self._font_cache:
   250→            for path in candidates:
   251→                try:
   252→                    self._font_cache[key] = ImageFont.truetype(path, size_px)
   253→                    break
   254→                except OSError:
   255→                    continue
   256→            else:
   257→                self._font_cache[key] = ImageFont.load_default()
   258→        return self._font_cache[key]
   259→
   260→    @staticmethod
   261→    def _tracked_text(
   262→        draw: Any, xy: tuple[int, int], text: str, font: Any, *, fill: tuple, tracking: int
   263→    ) -> None:
   264→        """Letter-spaced text (editorial eyebrow styling)."""
   265→        x, y = xy
   266→        for char in text:
   267→            draw.text((x, y), char, font=font, fill=fill)
   268→            x += draw.textlength(char, font=font) + tracking
   269→
   270→    # ------------------------------------------------------------------ apply
   271→    def apply(
   272→        self,
   273→        png: bytes,
   274→        *,
   275→        company_name: str,
   276→        domain: str | None,
   277→        vertical: str | None,
   278→        estimate: LeakageEstimate,
   279→        width: int,
   280→        height: int,
   281→    ) -> bytes:
   282→        from PIL import Image, ImageDraw, ImageFilter
   283→
   284→        base = Image.open(io.BytesIO(png)).convert("RGBA")
   285→        scale = base.width / self._settings.viewport_width
   286→        card_w, card_h = int(self.CARD_W * scale), int(self.CARD_H * scale)
   287→        margin = int(self.MARGIN * scale)
   288→        radius = int(self.RADIUS * scale)
   289→        x0, y0 = margin, max(0, base.height - card_h - margin)
   290→
   291→        # Floating-artifact shadow at ~10% opacity (DESIGN.md elevation spec).
   292→        shadow_layer = Image.new("RGBA", base.size, (0, 0, 0, 0))
   293→        ImageDraw.Draw(shadow_layer).rounded_rectangle(
   294→            [x0, y0 + int(16 * scale), x0 + card_w, y0 + card_h + int(16 * scale)],
   295→            radius=radius,
   296→            fill=_Tokens.SHADOW,
   297→        )
   298→        shadow_layer = shadow_layer.filter(ImageFilter.GaussianBlur(int(11 * scale)))
   299→        layer = shadow_layer
   300→
   301→        draw = ImageDraw.Draw(layer)
   302→        draw.rounded_rectangle(
   303→            [x0, y0, x0 + card_w, y0 + card_h],
   304→            radius=radius,
   305→            fill=_Tokens.PAPER,
   306→            outline=_Tokens.HAIRLINE,
   307→            width=max(1, int(scale)),
   308→        )
   309→
   310→        pad = int(24 * scale)
   311→        cx = x0 + pad
   312→        eyebrow_font = self._font(max(9, int(11 * scale)))
   313→        figure_font = self._font(max(20, int(38 * scale)), bold=True)
   314→        caption_font = self._font(max(9, int(13 * scale)))
   315→        meta_font = self._font(max(8, int(11 * scale)))
   316→        footer_font = self._font(max(8, int(10 * scale)))
   317→
   318→        # Eyebrow — letter-spaced category label (typographic tag, no badge).
   319→        self._tracked_text(
   320→            draw, (cx, y0 + int(20 * scale)), "SYNTROPIC // QUOTE LEAKAGE AUDIT",
   321→            eyebrow_font, fill=_Tokens.ASH, tracking=max(1, int(2 * scale)),
   322→        )
   323→
   324→        # Headline figure — the estimated annual net margin leakage.
   325→        figure = f"${estimate.annual_net_margin_leakage / 1e6:.2f}M"
   326→        draw.text((cx, y0 + int(44 * scale)), figure, font=figure_font, fill=_Tokens.INK)
   327→
   328→        # Caption.
   329→        draw.text(
   330→            (cx, y0 + int(96 * scale)), "estimated annual net margin leakage",
   331→            font=caption_font, fill=_Tokens.SLATE,
   332→        )
   333→
   334→        # Minimal gauge — mist track, ink fill, no chroma, no gridlines.
   335→        track_y = y0 + int(126 * scale)
   336→        track_h = max(3, int(6 * scale))
   337→        track_w = card_w - 2 * pad
   338→        draw.rounded_rectangle(
   339→            [cx, track_y, cx + track_w, track_y + track_h],
   340→            radius=track_h // 2, fill=_Tokens.MIST,
   341→        )
   342→        revenue_share = estimate.annual_net_margin_leakage / max(estimate.revenue_basis_usd, 1.0)
   343→        fill_frac = max(0.06, min(1.0, revenue_share / 0.035))  # gauge saturates at 3.5% of revenue
   344→        draw.rounded_rectangle(
   345→            [cx, track_y, cx + int(track_w * fill_frac), track_y + track_h],
   346→            radius=track_h // 2, fill=_Tokens.INK,
   347→        )
   348→
   349→        # Meta rows — the defensive math, in smoke gray.
   350→        vertical_label = _VERTICAL_LABELS.get((vertical or "").lower(), "Industrial Distribution")
   351→        draw.text(
   352→            (cx, track_y + int(16 * scale)),
   353→            f"≈ {revenue_share:.1%} of estimated revenue · {vertical_label}",
   354→            font=meta_font, fill=_Tokens.SMOKE,
   355→        )
   356→        draw.text(
   357→            (cx, track_y + int(34 * scale)),
   358→            f"{estimate.monthly_rfq_volume} RFQs/mo · "
   359→            f"{estimate.avg_quote_delay_hours:.0f}h avg quote turnaround",
   360→            font=meta_font, fill=_Tokens.SMOKE,
   361→        )
   362→
   363→        # Footer — attribution + audit deep link.
   364→        domain_label = (domain or "syntropic-os.com").lower()
   365→        draw.text(
   366→            (cx, y0 + card_h - int(30 * scale)),
   367→            f"{company_name} · {domain_label}/audit",
   368→            font=footer_font, fill=_Tokens.ASH,
   369→        )
   370→
   371→        composed = Image.alpha_composite(base, layer)
   372→        out = io.BytesIO()
   373→        composed.save(out, "PNG", optimize=True)
   374→        log.info(
   375→            "audit_widget_composed",
   376→            company=company_name,
   377→            figure=figure,
   378→            revenue_share=round(revenue_share, 5),
   379→            out_size=f"{composed.width}x{composed.height}",
   380→        )
   381→        return out.getvalue()

   384→# --------------------------------------------------------------------------- #
   385→# Asset storage — S3/R2 or local mounted dir
   386→# --------------------------------------------------------------------------- #
   387→
   388→
   389→class AssetStorage:
   390→    """Persists audit PNGs and returns their public CDN URL."""
   391→
   392→    def __init__(self, settings: AuditPreviewSettings | None = None) -> None:
   393→        self._settings = settings or get_settings().audit_preview
   394→        self._s3: Any | None = None
   395→
   396→    def _s3_client(self) -> Any:
   397→        if self._s3 is None:
   398→            import boto3  # lazy heavy import
   399→
   400→            secret = self._settings.aws_secret_access_key
   401→            self._s3 = boto3.client(
   402→                "s3",
   403→                region_name=self._settings.s3_region,
   404→                endpoint_url=self._settings.s3_endpoint_url,
   405→                aws_access_key_id=self._settings.aws_access_key_id,
   406→                aws_secret_access_key=secret.get_secret_value() if secret else None,
   407→            )
   408→        return self._s3
   409→
   410→    async def put_png(self, *, key: str, data: bytes) -> str:
   411→        backend = self._settings.storage_backend.lower()
   412→        if backend == "s3":
   413→            return await self._put_s3(key, data)
   414→        return await self._put_local(key, data)
   415→
   416→    async def _put_s3(self, key: str, data: bytes) -> str:
   417→        bucket = self._settings.s3_bucket
   418→        if not bucket:
   419→            raise AuditPreviewError("audit_preview.s3_bucket is required for the s3 backend")
   420→
   421→        def _upload() -> None:
   422→            self._s3_client().put_object(
   423→                Bucket=bucket,
   424→                Key=key,
   425→                Body=data,
   426→                ContentType="image/png",
   427→                CacheControl="public, max-age=31536000, immutable",
   428→            )
   429→
   430→        try:
   431→            await asyncio.to_thread(_upload)
   432→        except Exception as exc:  # noqa: BLE001 - surfaced as AuditPreviewError
   433→            raise AuditPreviewError(f"s3 upload failed for {key}: {exc}") from exc
   434→        url = f"{self._settings.asset_public_base_url}/{key}"
   435→        log.info("audit_asset_uploaded_s3", key=key, bucket=bucket)
   436→        return url
   437→
   438→    async def _put_local(self, key: str, data: bytes) -> str:
   439→        path = Path(self._settings.local_dir) / key
   440→
   441→        def _write() -> None:
   442→            path.parent.mkdir(parents=True, exist_ok=True)
   443→            path.write_bytes(data)
   444→
   445→        await asyncio.to_thread(_write)
   446→        url = f"{self._settings.asset_public_base_url}/{key}"
   447→        log.info("audit_asset_written_local", key=key, path=str(path))
   448→        return url

   455→
   456→class AuditPreviewGenerator:
   457→    """End-to-end: prospect row → screenshot → widget → CDN → DB + outbox."""
   458→
   459→    RFQ_PATH_CANDIDATES = ("/rfq", "/request-a-quote", "/request-quote", "/quote", "/contact", "/contact-us")
   460→
   461→    def __init__(
   462→        self,
   463→        db: Database,
   464→        outbox: OutboxStore,
   465→        *,
   466→        settings: AuditPreviewSettings | None = None,
   467→        scraping: ScrapingSettings | None = None,
   468→        capture: ScreenshotCapture | None = None,
   469→        overlay: AuditWidgetOverlay | None = None,
   470→        storage: AssetStorage | None = None,
   471→        http: httpx.AsyncClient | None = None,
   472→    ) -> None:
   473→        self._db = db
   474→        self._outbox = outbox
   475→        self._settings = settings or get_settings().audit_preview
   476→        self._scraping = scraping or get_settings().scraping
   477→        self._capture = capture or ScreenshotCapture(self._settings)
   478→        self._overlay = overlay or AuditWidgetOverlay(self._settings)
   479→        self._storage = storage or AssetStorage(self._settings)
   480→        self._http = http or httpx.AsyncClient(
   481→            headers={"User-Agent": self._scraping.user_agent},
   482→            timeout=httpx.Timeout(self._scraping.request_timeout_sec, connect=8.0),
   483→            follow_redirects=True,
   484→        )
   485→
   486→    async def aclose(self) -> None:
   487→        await self._http.aclose()
   488→
   489→    # ------------------------------------------------------------ public API
   490→    async def generate(self, prospect_id: str) -> AuditPreview:
   491→        import json as _json
   492→
   493→        row = await self._db.fetchrow(
   494→            """
   495→            SELECT id, name, domain, vertical, revenue_estimate_usd
   496→              FROM prospects WHERE id = $1
   497→            """,
   498→            uuid_mod.UUID(prospect_id),
   499→        )
   500→        if row is None:
   501→            raise AuditPreviewError("prospect not found", context={"prospect_id": prospect_id})
   502→        if not row["domain"]:
   503→            raise AuditPreviewError("prospect has no domain to capture", context={"prospect_id": prospect_id})
   504→
   505→        target_url = await self._resolve_capture_url(str(row["domain"]))
   506→        shot = await self._capture.capture(target_url)
   507→        estimate = build_leakage_estimate(
   508→            vertical=str(row["vertical"]) if row["vertical"] else None,
   509→            revenue_estimate_usd=row["revenue_estimate_usd"],
   510→        )
   511→        composed = self._overlay.apply(
   512→            shot.png,
   513→            company_name=str(row["name"]),
   514→            domain=str(row["domain"]),
   515→            vertical=str(row["vertical"]) if row["vertical"] else None,
   516→            estimate=estimate,
   517→            width=shot.width,
   518→            height=shot.height,
   519→        )
   520→        key = f"audit-previews/{row['id']}/{uuid_mod.uuid4().hex}.png"
   521→        public_url = await self._storage.put_png(key=key, data=composed)
   522→
   523→        preview = AuditPreview(
   524→            prospect_id=str(row["id"]),
   525→            source_url=target_url,
   526→            asset_key=key,
   527→            audit_image_url=public_url,
   528→            width=shot.width,
   529→            height=shot.height,
   530→            leakage=estimate,
   531→        )
   532→        await self._persist(
   533→            preview,
   534→            composed_payload=_json.dumps(preview.model_dump(mode="json"), default=str),
   535→        )
   536→        return preview
   537→
   538→    async def generate_batch(self, *, limit: int = 20) -> list[str]:
   539→        """Highest-scoring prospects without an audit image, bounded batch."""
   540→        rows = await self._db.fetch(
   541→            """
   542→            SELECT p.id FROM prospects p
   543→            LEFT JOIN LATERAL (
   544→                SELECT qlv_score FROM scores s
   545→                 WHERE s.prospect_id = p.id ORDER BY scored_at DESC LIMIT 1
   546→            ) latest ON TRUE
   547→             WHERE p.domain IS NOT NULL
   548→               AND p.audit_image_url IS NULL
   549→               AND p.ingestion <> 'filtered'
   550→             ORDER BY COALESCE(latest.qlv_score, 0) DESC, p.created_at
   551→             LIMIT $1
   552→            """,
   553→            limit,
   554→        )
   555→        done: list[str] = []
   556→        for row in rows:
   557→            try:
   558→                preview = await self.generate(str(row["id"]))
   559→                done.append(preview.prospect_id)
   560→            except Exception as exc:  # noqa: BLE001 - batch must survive failures
   561→                log.warning("audit_preview_batch_item_failed", prospect_id=str(row["id"]), error=str(exc))
   562→        return done
   563→

   565→    async def _resolve_capture_url(self, domain: str) -> str:
   566→        """Prefer the live RFQ page; fall back to the site root."""
   567→        base = f"https://{domain}"
   568→        for path in self.RFQ_PATH_CANDIDATES:
   569→            url = f"{base}{path}"
   570→            try:
   571→                response = await self._http.get(url)
   572→            except httpx.HTTPError as exc:
   573→                log.debug("audit_rfq_probe_failed", url=url, error=str(exc))
   574→                continue
   575→            if response.status_code == 200 and "html" in response.headers.get("content-type", ""):
   576→                return str(response.url)
   577→        log.info("audit_capture_fallback_homepage", domain=domain)
   578→        return base

   580→    async def _persist(self, preview: AuditPreview, *, composed_payload: str) -> None:
   581→        async with self._db.pool.acquire() as conn:
   582→            async with conn.transaction():
   583→                await conn.execute(
   584→                    """
   585→                    UPDATE prospects
   586→                       SET audit_image_url = $2, updated_at = now()
   587→                     WHERE id = $1
   588→                    """,
   589→                    uuid_mod.UUID(preview.prospect_id),
   590→                    preview.audit_image_url,
   591→                )
   592→                await self._outbox.enqueue(
   593→                    conn,
   594→                    aggregate_type="prospect",
   595→                    aggregate_id=preview.prospect_id,
   596→                    event_type="audit.preview_generated",
   597→                    payload={
   598→                        "audit_image_url": preview.audit_image_url,
   599→                        "asset_key": preview.asset_key,
   600→                        "source_url": preview.source_url,
   601→                        "annual_net_margin_leakage": preview.leakage.annual_net_margin_leakage,
   602→                        "methodology": preview.leakage.methodology,
   603→                    },
   604→                )
   605→        log.info(
   606→            "audit_preview_generated",
   607→            prospect_id=preview.prospect_id,
   608→            url=preview.audit_image_url,
   609→        )
