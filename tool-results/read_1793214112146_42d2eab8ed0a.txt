     1→"""Module 9 — Autonomous Closed-Loop Follow-Up Agent.
     2→
     3→A LangGraph state machine over inbound prospect replies, powered by
     4→Claude 3.5 Sonnet with forced tool schemas (same structured-output contract
     5→as ``vertebra.ai.scorer``).
     6→
     7→Loop
     8→----
     9→    inbound email → hygiene (strip quotes/signatures)
    10→                  → classify (Claude) ──► INTERESTED ──────► book urgent call on
    11→                  │                        │                 Josias' calendar
    12→                  │                        ├─► OBJECTION_* ─► draft AI response
    13→                  │                        │                 for human review
    14→                  │                        ├─► SEND_INFO ───► draft info response
    15→                  │                        │                 for human review
    16→                  │                        ├─► OUT_OF_OFFICE► reschedule dispatch
    17→                  │                        │                 past return date
    18→                  │                        └─► NOT_INTERESTED► suppress sequence
    19→                  └─► finalize: persist intent, drafts and outbox events
    20→
    21→Intent classes: ``interested`` · ``out_of_office`` · ``send_information`` ·
    22→``not_interested`` · ``objection_price`` · ``objection_erp`` · ``unknown``.
    23→
    24→Autonomy boundary
    25→-----------------
    26→Interested replies book calendar time immediately (the operator asked for
    27→speed); price/ERP objections and information requests are *drafted* and
    28→queued for human review in ``agent_drafts`` — Claude never sends a rebuttal
    29→on its own.
    30→
    31→Usage
    32→-----
    33→    python -m vertebra.ai.reply_agent --process-pending --limit 25
    34→    python -m vertebra.ai.reply_agent --reply-id <uuid>
    35→"""

    67→_QUOTE_MARKERS: tuple[re.Pattern[str], ...] = (
    68→    re.compile(r"^On .{0,200}wrote:\s*$", re.MULTILINE | re.IGNORECASE),
    69→    re.compile(r"^-+\s*Original Message\s*-+\s*$", re.MULTILINE | re.IGNORECASE),
    70→    re.compile(r"^From:\s.+$", re.MULTILINE),
    71→    re.compile(r"^Sent from my (iPhone|Android).*$", re.MULTILINE | re.IGNORECASE),
    72→)
    73→_SIGNATURE_SPLIT = re.compile(r"\n--\s*\n")
    74→
    75→
    76→def strip_quoted_text(body: str) -> str:
    77→    """Cut quoted history and signatures so the classifier sees the fresh text."""
    78→    cut = len(body)
    79→    signature = _SIGNATURE_SPLIT.search(body)
    80→    if signature:
    81→        cut = min(cut, signature.start())
    82→    for pattern in _QUOTE_MARKERS:
    83→        match = pattern.search(body)
    84→        if match:
    85→            cut = min(cut, match.start())
    86→    return body[:cut].strip()

   180→class ReplyClassifier:
   181→    """Claude 3.5 Sonnet classification with a forced tool schema."""
   182→
   183→    def __init__(self, settings: AnthropicSettings | None = None) -> None:
   184→        self._settings = settings or get_settings().anthropic
   185→        self._client: Any | None = None
   186→
   187→    def _anthropic(self) -> Any:
   188→        if self._client is None:
   189→            import anthropic  # lazy heavy import
   190→
   191→            self._client = anthropic.AsyncAnthropic(
   192→                api_key=self._settings.api_key.get_secret_value(),
   193→                max_retries=self._settings.max_retries,
   194→                timeout=self._settings.timeout_sec,
   195→            )
   196→        return self._client
   197→
   198→    async def classify(
   199→        self, *, lead_ctx: dict[str, Any], subject: str | None, body: str
   200→    ) -> tuple[ReplyIntent, float, str, ReplyExtraction]:
   201→        response = await self._anthropic().messages.create(
   202→            model=self._settings.model,
   203→            max_tokens=min(self._settings.max_tokens, 1200),
   204→            temperature=0.1,
   205→            system=_SYSTEM_CLASSIFY,
   206→            tools=[_CLASSIFY_TOOL],
   207→            tool_choice={"type": "tool", "name": _CLASSIFY_TOOL_NAME},
   208→            messages=[{"role": "user", "content": _classify_prompt(lead_ctx, subject, body)}],
   209→        )
   210→        for block in response.content:
   211→            if getattr(block, "type", None) == "tool_use" and block.name == _CLASSIFY_TOOL_NAME:
   212→                return self._parse(block.input)
   213→        raise ReplyAgentError(
   214→            "classifier returned no tool_use block", context={"model": self._settings.model}
   215→        )
   216→
   217→    def _parse(self, data: dict[str, Any]) -> tuple[ReplyIntent, float, str, ReplyExtraction]:
   218→        try:
   219→            intent = ReplyIntent(str(data["intent"]))
   220→        except (KeyError, ValueError) as exc:
   221→            raise ReplyAgentError(f"unknown intent from classifier: {data.get('intent')!r}") from exc
   222→        confidence = max(0.0, min(1.0, float(data.get("confidence", 0.0))))
   223→        rationale = str(data.get("rationale", ""))
   224→
   225→        return_date: datetime | None = None
   226→        raw_return = data.get("return_date")
   227→        if raw_return:
   228→            try:
   229→                return_date = datetime.fromisoformat(str(raw_return).replace("Z", "+00:00"))
   230→            except ValueError:
   231→                log.warning("classifier_return_date_unparseable", value=str(raw_return))
   232→        objection_type = data.get("objection_type")
   233→        if objection_type not in ("price", "erp", "other"):
   234→            objection_type = None
   235→        extraction = ReplyExtraction(
   236→            return_date=return_date,
   237→            objection_type=objection_type,  # type: ignore[arg-type]
   238→            requested_info=data.get("requested_info") or None,
   239→            meeting_readiness=bool(data.get("meeting_readiness", False)),
   240→            delegate_contact=data.get("delegate_contact") or None,
   241→        )
   242→        return intent, confidence, rationale, extraction

   245→class DraftWriter:
   246→    """Claude 3.5 Sonnet draft generation for objection / info replies."""
   247→
   248→    def __init__(
   249→        self,
   250→        *,
   251→        settings: AnthropicSettings | None = None,
   252→        agent_settings: ReplyAgentSettings | None = None,
   253→    ) -> None:
   254→        self._settings = settings or get_settings().anthropic
   255→        self._agent = agent_settings or get_settings().reply_agent
   256→        self._client: Any | None = None
   257→
   258→    def _anthropic(self) -> Any:
   259→        if self._client is None:
   260→            import anthropic  # lazy heavy import
   261→
   262→            self._client = anthropic.AsyncAnthropic(
   263→                api_key=self._settings.api_key.get_secret_value(),
   264→                max_retries=self._settings.max_retries,
   265→                timeout=self._settings.timeout_sec,
   266→            )
   267→        return self._client
   268→
   269→    async def draft(
   270→        self, *, lead_ctx: dict[str, Any], intent: ReplyIntent, reply_body: str
   271→    ) -> tuple[str, str]:
   272→        user = (
   273→            f"{_lead_block(lead_ctx)}\n\n"
   274→            f"Detected intent: {intent.value}\n\n"
   275→            f"Prospect's reply:\n\"\"\"\n{reply_body[:2000]}\n\"\"\"\n\n"
   276→            f"Draft the reply now. Hard limit: {self._agent.draft_max_words} words."
   277→        )
   278→        response = await self._anthropic().messages.create(
   279→            model=self._settings.model,
   280→            max_tokens=min(self._settings.max_tokens, 1400),
   281→            temperature=0.3,
   282→            system=_SYSTEM_DRAFT,
   283→            tools=[_DRAFT_TOOL],
   284→            tool_choice={"type": "tool", "name": _DRAFT_TOOL_NAME},
   285→            messages=[{"role": "user", "content": user}],
   286→        )
   287→        for block in response.content:
   288→            if getattr(block, "type", None) == "tool_use" and block.name == _DRAFT_TOOL_NAME:
   289→                subject = str(block.input.get("subject", "Re: your note")).strip()
   290→                body = str(block.input.get("body", "")).strip()
   291→                body = self._enforce_word_cap(body)
   292→                if len(body) < 40:
   293→                    raise ReplyAgentError("draft rejected: body too short", context={"intent": intent.value})
   294→                return subject, body
   295→        raise ReplyAgentError("drafter returned no tool_use block", context={"intent": intent.value})
   296→
   297→    def _enforce_word_cap(self, body: str) -> str:
   298→        cap = self._agent.draft_max_words
   299→        words = body.split()
   300→        if len(words) <= cap:
   301→            return body
   302→        log.warning("draft_word_cap_enforced", limit=cap, actual=len(words))
   303→        return " ".join(words[:cap]).rstrip(",;") + "…"
   304→
   305→
   306→def _lead_block(lead_ctx: dict[str, Any]) -> str:
   307→    revenue = lead_ctx.get("revenue_estimate_usd")
   308→    revenue_line = f"${revenue:,.0f}" if revenue else "unknown"
   309→    lines = [
   310→        f"Company: {lead_ctx.get('name') or 'unknown'}",
   311→        f"Vertical: {lead_ctx.get('vertical') or 'unknown'}",
   312→        f"Location: {lead_ctx.get('city') or 'unknown'}, {lead_ctx.get('country') or 'unknown'}",
   313→        f"Revenue estimate: {revenue_line}",
   314→        f"Detected ERP: {lead_ctx.get('detected_erp') or 'none detected'}",
   315→        f"Quote portal type: {lead_ctx.get('quote_portal_type') or 'unknown'}",
   316→        f"QLV score: {lead_ctx.get('qlv_score') or 'unscored'}",
   317→    ]
   318→    contact = lead_ctx.get("contact_name")
   319→    if contact:
   320→        lines.insert(1, f"Contact: {contact}")
   321→    rationale = lead_ctx.get("score_rationale")
   322→    if rationale:
   323→        lines.append(f"Why they were pursued (excerpt): {str(rationale)[:300]}")
   324→    return "\n".join(lines)
   325→
   326→
   327→def _classify_prompt(lead_ctx: dict[str, Any], subject: str | None, body: str) -> str:
   328→    return (
   329→        f"{_lead_block(lead_ctx)}\n\n"
   330→        f"Subject: {subject or '(none)'}\n\n"
   331→        f"Reply body:\n\"\"\"\n{body[:3000]}\n\"\"\"\n\n"
   332→        "Classify this reply now."
   333→    )

   340→
   341→class ReplyGraphState(TypedDict, total=False):
   342→    """Single shared state flowing through the agent loop."""
   343→
   344→    reply_id: str
   345→    prospect_id: str | None
   346→    from_email: str
   347→    subject: str
   348→    body: str
   349→    lead_ctx: dict[str, Any]
   350→    intent: str
   351→    confidence: float
   352→    rationale: str
   353→    extraction: dict[str, Any]
   354→    actions: Annotated[list[str], lambda a, b: a + b]  # reducer: append-only
   355→    draft_subject: str
   356→    draft_body: str
   357→    calendar_event_id: str
   358→    calendar_event_link: str
   359→    rescheduled_to: str
   360→    review_required: bool
   361→    error: str
   362→
   363→
   364→def build_reply_graph(
   365→    *,
   366→    db: Database,
   367→    outbox: OutboxStore,
   368→    classifier: ReplyClassifier,
   369→    drafter: DraftWriter,
   370→    calendar: CalendarSync,
   371→    settings: ReplyAgentSettings | None = None,
   372→):
   373→    """Compile the classify → act → finalize state machine."""
   374→    agent_settings = settings or get_settings().reply_agent
   375→
   376→    # --------------------------------------------------------------- classify
   377→    async def classify_node(state: ReplyGraphState) -> dict[str, Any]:
   378→        intent, confidence, rationale, extraction = await classifier.classify(
   379→            lead_ctx=state["lead_ctx"], subject=state.get("subject"), body=state["body"]
   380→        )
   381→        return {
   382→            "intent": intent.value,
   383→            "confidence": confidence,
   384→            "rationale": rationale,
   385→            "extraction": json.loads(extraction.model_dump_json(exclude_none=True)),
   386→        }
   387→
   388→    # -------------------------------------------------------------- book call
   389→    async def book_call_node(state: ReplyGraphState) -> dict[str, Any]:
   390→        lead = state["lead_ctx"]
   391→        try:
   392→            created = await calendar.create_urgent_call_invite(
   393→                company_name=str(lead.get("name") or "unknown company"),
   394→                contact_name=str(lead.get("contact_name") or state["from_email"]),
   395→                prospect_id=str(lead.get("id") or state.get("prospect_id") or "unsynced"),
   396→                intent=state["intent"],
   397→                reply_excerpt=state["body"][:400],
   398→                qlv_score=lead.get("qlv_score"),
   399→                audit_image_url=lead.get("audit_image_url"),
   400→            )
   401→        except Exception as exc:  # noqa: BLE001 - calendar outage must not kill the loop
   402→            log.warning("urgent_call_booking_failed", error=str(exc))
   403→            return {"error": f"calendar_booking_failed: {exc}"}
   404→        return {
   405→            "calendar_event_id": created["event_id"],
   406→            "calendar_event_link": created.get("html_link"),
   407→        }
   408→
   409→    # ---------------------------------------------------------- draft replies
   410→    async def draft_response_node(state: ReplyGraphState) -> dict[str, Any]:
   411→        subject, body = await drafter.draft(
   412→            lead_ctx=state["lead_ctx"],
   413→            intent=ReplyIntent(state["intent"]),
   414→            reply_body=state["body"],
   415→        )
   416→        return {
   417→            "draft_subject": subject,
   418→            "draft_body": body,
   419→            "review_required": True,
   420→        }
   421→
   422→    # -------------------------------------------------------------- reschedule
   423→    async def reschedule_node(state: ReplyGraphState) -> dict[str, Any]:
   424→        extraction = state.get("extraction") or {}
   425→        return_date = extraction.get("return_date")
   426→        try:
   427→            target = (
   428→                datetime.fromisoformat(str(return_date))
   429→                if return_date
   430→                else utcnow() + timedelta(days=7)
   431→            )
   432→        except ValueError:
   433→            target = utcnow() + timedelta(days=7)
   434→        if target.tzinfo is None:
   435→            target = target.replace(tzinfo=timezone.utc)
   436→        target += timedelta(hours=agent_settings.ooo_buffer_hours)
   437→
   438→        prospect_uuid = uuid_mod.UUID(state["prospect_id"]) if state.get("prospect_id") else None
   439→        if prospect_uuid is not None:
   440→            async with db.pool.acquire() as conn:
   441→                async with conn.transaction():
   442→                    await conn.execute(
   443→                        """
   444→                        UPDATE dispatches
   445→                           SET send_after = GREATEST(COALESCE(send_after, $2), $2),
   446→                               updated_at = now()
   447→                         WHERE prospect_id = $1 AND status = 'queued'
   448→                        """,
   449→                        prospect_uuid,
   450→                        target,
   451→                    )
   452→                    await outbox.enqueue(
   453→                        conn,
   454→                        aggregate_type="reply",
   455→                        aggregate_id=state["reply_id"],
   456→                        event_type="reply.ooo_rescheduled",
   457→                        payload={"rescheduled_to": target.isoformat()},
   458→                    )
   459→        return {"rescheduled_to": target.isoformat()}
   460→
   461→    # ---------------------------------------------------------------- suppress
   462→    async def suppress_node(state: ReplyGraphState) -> dict[str, Any]:
   463→        prospect_uuid = uuid_mod.UUID(state["prospect_id"]) if state.get("prospect_id") else None
   464→        if prospect_uuid is not None:
   465→            async with db.pool.acquire() as conn:
   466→                async with conn.transaction():
   467→                    await conn.execute(
   468→                        """
   469→                        UPDATE dispatches
   470→                           SET status = 'suppressed', updated_at = now()
   471→                         WHERE prospect_id = $1 AND status = 'queued'
   472→                        """,
   473→                        prospect_uuid,
   474→                    )
   475→                    await outbox.enqueue(
   476→                        conn,
   477→                        aggregate_type="reply",
   478→                        aggregate_id=state["reply_id"],
   479→                        event_type="reply.sequence_suppressed",
   480→                        payload={"reason": "not_interested"},
   481→                    )
   482→        return {}
   483→
   484→    # ---------------------------------------------------------------- finalize
   485→    async def finalize_node(state: ReplyGraphState) -> dict[str, Any]:
   486→        """Persist the decision atomically: reply row, draft, outbox events."""
   487→        actions: list[str] = []
   488→        if state.get("calendar_event_id"):
   489→            actions.append(AgentActionType.BOOK_URGENT_CALL.value)
   490→        if state.get("draft_body"):
   491→            actions.append(AgentActionType.DRAFT_FOR_REVIEW.value)
   492→        if state.get("rescheduled_to"):
   493→            actions.append(AgentActionType.RESCHEDULE_SEQUENCE.value)
   494→        if "not_interested" == state.get("intent"):
   495→            actions.append(AgentActionType.SUPPRESS_SEQUENCE.value)
   496→        if state.get("error"):
   497→            actions.append(AgentActionType.ESCALATE_HUMAN.value)
   498→
   499→        async with db.pool.acquire() as conn:
   500→            async with conn.transaction():
   501→                await conn.execute(
   502→                    """
   503→                    UPDATE inbound_replies
   504→                       SET intent = $2, intent_confidence = $3, rationale = $4,
   505→                           extracted = $5::jsonb, processed_at = now()
   506→                     WHERE id = $1
   507→                    """,
   508→                    uuid_mod.UUID(state["reply_id"]),
   509→                    state["intent"],
   510→                    state.get("confidence"),
   511→                    state.get("rationale"),
   512→                    json.dumps(state.get("extraction") or {}, default=str),
   513→                )
   514→                if state.get("draft_body"):
   515→                    await conn.execute(
   516→                        """
   517→                        INSERT INTO agent_drafts (reply_id, prospect_id, intent, subject, body)
   518→                        VALUES ($1, $2, $3, $4, $5)
   519→                        """,
   520→                        uuid_mod.UUID(state["reply_id"]),
   521→                        uuid_mod.UUID(state["prospect_id"]) if state.get("prospect_id") else None,
   522→                        state["intent"],
   523→                        state.get("draft_subject") or "(no subject)",
   524→                        state["draft_body"],
   525→                    )
   526→                await outbox.enqueue(
   527→                    conn,
   528→                    aggregate_type="reply",
   529→                    aggregate_id=state["reply_id"],
   530→                    event_type="reply.classified",
   531→                    payload={
   532→                        "prospect_id": state.get("prospect_id"),
   533→                        "intent": state["intent"],
   534→                        "confidence": state.get("confidence"),
   535→                        "actions": actions,
   536→                        "review_required": bool(state.get("draft_body")),
   537→                        "calendar_event_id": state.get("calendar_event_id"),
   538→                        "rescheduled_to": state.get("rescheduled_to"),
   539→                        "error": state.get("error"),
   540→                    },
   541→                )
   542→        log.info(
   543→            "reply_finalized",
   544→            reply_id=state["reply_id"],
   545→            intent=state["intent"],
   546→            actions=actions,
   547→        )
   548→        state["actions"] = actions  # surface on the final state for AgentDecision
   549→        return {"actions": actions}
   550→
   551→    def _route(state: ReplyGraphState) -> str:
   552→        return {
   553→            ReplyIntent.INTERESTED.value: "book_call",
   554→            ReplyIntent.OBJECTION_PRICE.value: "draft_response",
   555→            ReplyIntent.OBJECTION_ERP.value: "draft_response",
   556→            ReplyIntent.SEND_INFORMATION.value: "draft_response",
   557→            ReplyIntent.OUT_OF_OFFICE.value: "reschedule",
   558→            ReplyIntent.NOT_INTERESTED.value: "suppress",
   559→        }.get(state.get("intent", ""), "finalize")
   560→
   561→    graph = StateGraph(ReplyGraphState)
   562→    graph.add_node("classify", classify_node)
   563→    graph.add_node("book_call", book_call_node)
   564→    graph.add_node("draft_response", draft_response_node)
   565→    graph.add_node("reschedule", reschedule_node)
   566→    graph.add_node("suppress", suppress_node)
   567→    graph.add_node("finalize", finalize_node)
   568→
   569→    graph.add_edge(START, "classify")
   570→    graph.add_conditional_edges(
   571→        "classify",
   572→        _route,
   573→        {
   574→            "book_call": "book_call",
   575→            "draft_response": "draft_response",
   576→            "reschedule": "reschedule",
   577→            "suppress": "suppress",
   578→            "finalize": "finalize",
   579→        },
   580→    )
   581→    for node in ("book_call", "draft_response", "reschedule", "suppress"):
   582→        graph.add_edge(node, "finalize")
   583→    graph.add_edge("finalize", END)
   584→    return graph.compile()
   585→

   588→# Orchestrator
   589→# --------------------------------------------------------------------------- #
   590→
   591→_LEAD_CONTEXT_SQL = """
   592→SELECT p.id, p.name, p.domain, p.vertical, p.city, p.state_province, p.country,
   593→       p.revenue_estimate_usd, p.detected_erp, p.quote_portal_type, p.audit_image_url,
   594→       COALESCE(latest.qlv_score, 0) AS qlv_score,
   595→       latest.rationale AS score_rationale,
   596→       (c.first_name || ' ' || c.last_name) AS contact_name
   597→  FROM prospects p
   598→  LEFT JOIN LATERAL (
   599→      SELECT qlv_score, rationale FROM scores s
   600→       WHERE s.prospect_id = p.id ORDER BY scored_at DESC LIMIT 1
   601→  ) latest ON TRUE
   602→  LEFT JOIN LATERAL (
   603→      SELECT first_name, last_name FROM contacts
   604→       WHERE prospect_id = p.id ORDER BY created_at LIMIT 1
   605→  ) c ON TRUE
   606→ WHERE p.id = $1
   607→"""
   608→
   609→_RESOLVE_PROSPECT_SQL = """
   610→WITH by_email AS (
   611→    SELECT id FROM prospects WHERE lower(best_email) = lower($1) LIMIT 1
   612→),
   613→by_domain AS (
   614→    SELECT p.id FROM prospects p
   615→     WHERE p.domain = $2
   616→       AND NOT EXISTS (SELECT 1 FROM by_email)
   617→     ORDER BY p.created_at LIMIT 1
   618→)
   619→SELECT id FROM by_email
   620→UNION ALL
   621→SELECT id FROM by_domain
   622→LIMIT 1
   623→"""

   626→class ReplyAgent:
   627→    """Registers inbound replies, runs the graph and returns decisions."""
   628→
   629→    def __init__(
   630→        self,
   631→        db: Database,
   632→        outbox: OutboxStore,
   633→        *,
   634→        classifier: ReplyClassifier | None = None,
   635→        drafter: DraftWriter | None = None,
   636→        calendar: CalendarSync | None = None,
   637→        settings: ReplyAgentSettings | None = None,
   638→    ) -> None:
   639→        self._db = db
   640→        self._outbox = outbox
   641→        self._classifier = classifier or ReplyClassifier()
   642→        self._drafter = drafter or DraftWriter()
   643→        self._calendar = calendar or CalendarSync()
   644→        self._settings = settings or get_settings().reply_agent
   645→        self._graph = build_reply_graph(
   646→            db=db,
   647→            outbox=outbox,
   648→            classifier=self._classifier,
   649→            drafter=self._drafter,
   650→            calendar=self._calendar,
   651→            settings=self._settings,
   652→        )
   653→
   654→    # ------------------------------------------------------------- public API
   655→    async def register_inbound(
   656→        self,
   657→        *,
   658→        from_email: str,
   659→        subject: str | None,
   660→        body: str,
   661→        message_id: str | None = None,
   662→        prospect_id: str | None = None,
   663→        received_at: datetime | None = None,
   664→    ) -> str:
   665→        """Persist one inbound reply (hygiene-cleaned) and return its id."""
   666→        cleaned = strip_quoted_text(body)
   667→        if not cleaned:
   668→            raise ReplyAgentError("reply body is empty after stripping quoted text")
   669→        resolved = prospect_id or await self._resolve_prospect(from_email)
   670→        row = await self._db.fetchrow(
   671→            """
   672→            INSERT INTO inbound_replies (prospect_id, message_id, from_email, subject, body, received_at)
   673→            VALUES ($1, $2, $3, $4, $5, COALESCE($6, now()))
   674→            RETURNING id
   675→            """,
   676→            uuid_mod.UUID(resolved) if resolved else None,
   677→            message_id,
   678→            from_email,
   679→            subject,
   680→            cleaned,
   681→            received_at,
   682→        )
   683→        log.info(
   684→            "inbound_reply_registered",
   685→            reply_id=str(row["id"]),
   686→            from_email=from_email,
   687→            prospect_id=resolved,
   688→        )
   689→        return str(row["id"])
   690→
   691→    async def process_reply(self, reply_id: str) -> AgentDecision:
   692→        """Run the full agent loop for one registered reply."""
   693→        row = await self._db.fetchrow(
   694→            "SELECT id, prospect_id, from_email, subject, body FROM inbound_replies WHERE id = $1",
   695→            uuid_mod.UUID(reply_id),
   696→        )
   697→        if row is None:
   698→            raise ReplyAgentError("reply not found", context={"reply_id": reply_id})
   699→
   700→        lead_ctx = await self._lead_context(row["prospect_id"])
   701→        if lead_ctx is None:
   702→            domain = str(row["from_email"]).rsplit("@", 1)[-1].lower()
   703→            lead_ctx = {"name": domain, "domain": domain}
   704→
   705→        final_state: dict[str, Any] = await self._graph.ainvoke(
   706→            {
   707→                "reply_id": str(row["id"]),
   708→                "prospect_id": str(row["prospect_id"]) if row["prospect_id"] else None,
   709→                "from_email": str(row["from_email"]),
   710→                "subject": str(row["subject"] or ""),
   711→                "body": str(row["body"]),
   712→                "lead_ctx": lead_ctx,
   713→            }
   714→        )
   715→        return AgentDecision(
   716→            reply_id=str(row["id"]),
   717→            prospect_id=str(row["prospect_id"]) if row["prospect_id"] else None,
   718→            intent=ReplyIntent(final_state.get("intent", ReplyIntent.UNKNOWN.value)),
   719→            confidence=float(final_state.get("confidence", 0.0)),
   720→            rationale=str(final_state.get("rationale", "")),
   721→            actions=[AgentActionType(a) for a in final_state.get("actions", [])],
   722→            draft_subject=final_state.get("draft_subject"),
   723→            draft_body=final_state.get("draft_body"),
   724→            calendar_event_id=final_state.get("calendar_event_id"),
   725→            calendar_event_link=final_state.get("calendar_event_link"),
   726→            rescheduled_to=final_state.get("rescheduled_to"),
   727→            review_required=bool(final_state.get("draft_body")),
   728→        )
   729→
   730→    async def process_pending(self, *, limit: int | None = None) -> list[AgentDecision]:
   731→        """Drain unprocessed replies oldest-first (worker/CLI entrypoint)."""
   732→        rows = await self._db.fetch(
   733→            """
   734→            SELECT id FROM inbound_replies
   735→             WHERE processed_at IS NULL
   736→             ORDER BY received_at
   737→             LIMIT $1
   738→            """,
   739→            limit or self._settings.process_pending_limit,
   740→        )
   741→        decisions: list[AgentDecision] = []
   742→        for row in rows:
   743→            try:
   744→                decisions.append(await self.process_reply(str(row["id"])))
   745→            except Exception as exc:  # noqa: BLE001 - batch must survive bad replies
   746→                log.exception("reply_processing_failed", reply_id=str(row["id"]), error=str(exc))
   747→        return decisions
   748→
   749→    # -------------------------------------------------------------- internals
   750→    async def _resolve_prospect(self, from_email: str) -> str | None:
   751→        domain = from_email.rsplit("@", 1)[-1].lower()
   752→        row = await self._db.fetchrow(_RESOLVE_PROSPECT_SQL, from_email, domain)
   753→        return str(row["id"]) if row else None
   754→
   755→    async def _lead_context(self, prospect_id: str | None) -> dict[str, Any] | None:
   756→        if not prospect_id:
   757→            return None
   758→        row = await self._db.fetchrow(_LEAD_CONTEXT_SQL, uuid_mod.UUID(prospect_id))
   759→        if row is None:
   760→            return None
   761→        return {
   762→            "id": str(row["id"]),
   763→            "name": row["name"],
   764→            "domain": row["domain"],
   765→            "vertical": row["vertical"],
   766→            "city": row["city"],
   767→            "state_province": row["state_province"],
   768→            "country": row["country"],
   769→            "revenue_estimate_usd": row["revenue_estimate_usd"],
   770→            "detected_erp": row["detected_erp"],
   771→            "quote_portal_type": row["quote_portal_type"],
   772→            "audit_image_url": row["audit_image_url"],
   773→            "qlv_score": row["qlv_score"],
   774→            "score_rationale": row["score_rationale"],
   775→            "contact_name": (row["contact_name"] or "").strip() or None,
   776→        }

   784→async def _run(args: list[str]) -> int:
   785→    from vertebra.logging_conf import configure_logging
   786→
   787→    settings = get_settings()
   788→    configure_logging(settings.log_level, settings.log_json)
   789→    db = Database(settings.database)
   790→    await db.connect()
   791→    outbox = OutboxStore(db)
   792→    agent = ReplyAgent(db, outbox)
   793→    try:
   794→        if args.reply_id:
   795→            decision = await agent.process_reply(args.reply_id)
   796→            print(json.dumps(decision.model_dump(mode="json"), indent=2, default=str))
   797→        else:
   798→            decisions = await agent.process_pending(limit=args.limit)
   799→            print(json.dumps([d.model_dump(mode="json") for d in decisions], indent=2, default=str))
   800→    finally:
   801→        await db.close()
   802→    return 0
