     7→ * at US & Canada industrial distributors.
     8→ */

    34→
    35→interface TelemetryPush {
    36→  event: {
    37→    id: string;

    41→    eventType: string;
    42→    url: string | null;
    43→    city: string | null;

    45→    device: string | null;
    46→    durationMs: number | null;

    51→}

    55→    replyId: string;

    59→    intent: string;

    64→    receivedAtMs: number;

    66→}

    69→  id: string;
    70→  kind: "alert" | "info";
    71→  title: string;

    74→}

    76→const VIEW_TITLES: Record<ViewKey, string> = {
    77→  radar: "Radar — margin telemetry, live",
    78→  strike: "Strike — recovery queue",
    79→  theatre: "Theatre — spatial cluster board",
    80→  pipeline: "Pipeline — machine state & triage",
    81→  portal: "Portal — margin defense entry",
    82→  ops: "Ops — engine health",
    83→};
    84→
    85→const VIEW_KEYS: ViewKey[] = ["radar", "strike", "theatre", "pipeline", "portal", "ops"];
    86→
    87→export function CommandCentre() {
    88→  const [snap, setSnap] = useState<CommandSnapshot | null>(null);
    89→  const [view, setView] = useState<ViewKey>("radar");
    90→  const [persona, setPersona] = useState<PersonaId>("coo");
    91→  const [dossier, setDossier] = useState<CommandLead | null>(null);
    92→  const [dossierOpen, setDossierOpen] = useState(false);
    93→  const [cursorId, setCursorId] = useState<string | null>(null);
    94→  const [filter, setFilter] = useState<StrikeFilter>("all");
    95→  const [toasts, setToasts] = useState<Toast[]>([]);
    96→  const [connected, setConnected] = useState(false);
    97→  const [clock, setClock] = useState("");
    98→  const [fresh, setFresh] = useState<Map<string, number>>(new Map());
    99→  const [paletteOpen, setPaletteOpen] = useState(false);
   100→  const [now, setNow] = useState(() => Date.now());
   101→  const socketRef = useRef<Socket | null>(null);
   102→  const alertSeen = useRef<Set<string>>(new Set());
   103→  const bootAlerted = useRef(false);
   104→
   105→  // ---- persona persistence ----------------------------------------------- //
   106→  useEffect(() => {
   107→    const saved = window.localStorage.getItem("syntropic:persona");
   108→    if (saved && PERSONAS.some((p) => p.id === saved)) setPersona(saved as PersonaId);
   109→  }, []);
   110→  const onPersona = useCallback((p: PersonaId) => {
   111→    setPersona(p);
   112→    window.localStorage.setItem("syntropic:persona", p);
   113→  }, []);
   114→
   115→  // ---- clock tick -------------------------------------------------------- //
   116→  useEffect(() => {
   117→    const t = setInterval(() => {
   118→      setNow(Date.now());
   119→      setClock(
   120→        new Intl.DateTimeFormat(undefined, { hour: "2-digit", minute: "2-digit", second: "2-digit", hour12: false }).format(new Date()),
   121→      );
   122→    }, 1000);
   123→    return () => clearInterval(t);
   124→  }, []);
   125→
   126→  // ---- snapshot loading -------------------------------------------------- //
   127→  const refresh = useCallback(async () => {
   128→    try {
   129→      const res = await fetch("/api/command", { cache: "no-store" });
   130→      if (!res.ok) return;
   131→      const data = (await res.json()) as CommandSnapshot;
   132→      setSnap(data);
   133→      if (!bootAlerted.current) {
   134→        bootAlerted.current = true;
   135→        const recent = data.alerts.filter((a) => Date.now() - a.createdAtMs <= 15 * 60_000).slice(0, 2);
   136→        for (const a of recent) {
   137→          const key = `boot:${a.id}`;
   138→          if (alertSeen.current.has(key)) continue;
   139→          alertSeen.current.add(key);
   140→          pushToast({ kind: "alert", title: a.title, body: a.body, prospectId: a.prospectId });
   141→          chime("alert");
   142→        }
   143→      }
   144→    } catch {
   145→      /* offline — retry on next tick */
   146→    }
   147→  }, []);
   148→
   149→  const pushToast = useCallback((t: Omit<Toast, "id">) => {
   150→    const id = `${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
   151→    setToasts((prev) => [{ id, ...t }, ...prev].slice(0, 4));
   152→    setTimeout(() => setToasts((prev) => prev.filter((x) => x.id !== id)), 10_000);
   153→  }, []);
   154→
   155→  useEffect(() => {
   156→    void refresh();
   157→    const t = setInterval(() => void refresh(), 60_000);
   158→    return () => clearInterval(t);
   159→  }, [refresh]);
   160→
   161→  // ---- socket bridge ----------------------------------------------------- //
   162→  useEffect(() => {
   163→    const socket = io("/?XTransformPort=3003", {
   164→      path: "/",
   165→      transports: ["websocket", "polling"],
   166→      reconnection: true,
   167→      reconnectionDelay: 2000,
   168→      timeout: 10_000,
   169→      forceNew: true,
   170→    });
   171→    socketRef.current = socket;
   172→
   173→    socket.on("connect", () => setConnected(true));
   174→    socket.on("disconnect", () => setConnected(false));
   175→
   176→    socket.on("telemetry", (push: TelemetryPush) => {
   177→      const ev = push.event;
   178→      const atMs = ev.occurredAtMs || Date.now();
   179→      setFresh((prev) => new Map(prev).set(ev.id, Date.now() + 90_000));
   180→
   181→      setSnap((prev) => {
   182→        if (!prev) return prev;
   183→        const highSignal = isHighSignal(ev.eventType);
   184→        const leads = prev.leads.map((lead) => {
   185→          if (lead.id !== ev.prospectId) return lead;
   186→          const telemetry = {
   187→            lastEventType: ev.eventType,
   188→            lastEventAtMs: atMs,
   189→            lastHighSignalAtMs: highSignal ? atMs : lead.telemetry.lastHighSignalAtMs,
   190→            events48h: lead.telemetry.events48h + 1,
   191→            totalSessionMs: lead.telemetry.totalSessionMs + (ev.durationMs ?? 0),
   192→          };
   193→          const priority =
   194→            effectivePriority(
   195→              { qlvScore: lead.score?.qlvScore ?? null, lastHighSignalAtMs: telemetry.lastHighSignalAtMs, lastEventType: telemetry.lastEventType },
   196→              atMs,
   197→            ) * (lead.__called ? 0.5 : 1);
   198→          return {
   199→            ...lead,
   200→            telemetry,
   201→            lastIntentHeat: ev.lastIntentHeat ?? lead.lastIntentHeat,
   202→            priority,
   203→            highIntentNow:
   204→              highSignal && telemetry.lastHighSignalAtMs != null && (lead.score?.qlvScore ?? 0) >= 85 && atMs - telemetry.lastHighSignalAtMs <= 15 * 60_000,
   205→          };
   206→        });
   207→        const source = prev.leads.find((l) => l.id === ev.prospectId);
   208→        const feed = [
   209→          {
   210→            id: ev.id,
   211→            prospectId: ev.prospectId,
   212→            prospectName: ev.prospectName,
   213→            qlvScore: ev.qlvScore,
   214→            clusterName: source?.clusterName ?? null,
   215→            vertical: source?.vertical ?? null,
   216→            eventType: ev.eventType,
   217→            url: ev.url,
   218→            city: ev.city,
   219→            country: ev.country,
   220→            device: ev.device,
   221→            durationMs: ev.durationMs,
   222→            occurredAtMs: atMs,
   223→            alertFired: Boolean(push.alert),
   224→          },
   225→          ...prev.feed,
   226→        ].slice(0, 40);
   227→        return { ...prev, leads, feed };
   228→      });
   229→
   230→      const score = ev.qlvScore ?? 0;
   231→      if (push.alert || (isHighSignal(ev.eventType) && score >= 85)) {
   232→        const dedupe = `${ev.prospectId}:${ev.eventType}:${Math.floor(atMs / 900_000)}`;
   233→        if (!alertSeen.current.has(dedupe)) {
   234→          alertSeen.current.add(dedupe);
   235→          pushToast({
   236→            kind: "alert",
   237→            title: push.alert?.title ?? "HIGH-INTENT PROSPECT ON PORTAL",
   238→            body: push.alert?.body ?? [
   239→              `Company: ${ev.prospectName}`,
   240→              `QLV: ${score}/100`,
   241→              `Location: ${ev.city ?? "—"}, ${ev.country ?? "—"}`,
   242→              `Action: Call direct line immediately.`,
   243→            ].join("\n"),
   244→            prospectId: ev.prospectId,
   245→          });
   246→          chime("alert");
   247→        }
   248→      } else if (score >= 70 && isHighSignal(ev.eventType)) {
   249→        pushToast({
   250→          kind: "info",
   251→          title: `${ev.eventType.replace(/_/g, " ")} · QLV ${score}`,
   252→          body: `${ev.prospectName} — ${ev.city ?? "—"}, ${ev.country ?? "—"}`,
   253→          prospectId: ev.prospectId,
   254→        });
   255→      }
   256→    });
   257→
   258→    socket.on("reply", (push: ReplyPush) => {
   259→      const r = push.reply;
   260→      pushToast({
   261→        kind: r.tier === 2 ? "alert" : "info",
   262→        title: r.tier === 2 ? `Reply agent · draft awaiting review` : `Reply agent · auto-resolved`,
   263→        body: [
   264→          `${r.prospectName} — QLV ${r.qlvScore ?? "—"}`,
   265→          `${INTENT_LABELS[r.intent as keyof typeof INTENT_LABELS] ?? r.intent} (${Math.round((r.confidence ?? 0) * 100)}% conf)`,
   266→          r.tier === 2 ? `Action: review the draft in Pipeline.` : `Action taken: ${r.action.replace(/_/g, " ")}.`,
   267→        ].join("\n"),
   268→        prospectId: r.prospectId,
   269→      });
   270→      chime(r.tier === 2 ? "alert" : "info");
   271→      void refresh();
   272→    });
   273→
   274→    return () => {
   275→      socket.removeAllListeners();
   276→      socket.close();
   277→      socketRef.current = null;
   278→    };
   279→  }, [pushToast, refresh]);
   280→
   281→  // ---- stale fresh-event cleanup ---------------------------------------- //
   282→  useEffect(() => {
   283→    const t = setInterval(() => {
   284→      setFresh((prev) => {
   285→        const next = new Map(prev);
   286→        for (const [k, until] of next) if (until < Date.now()) next.delete(k);
   287→        return next.size === prev.size ? prev : next;
   288→      });
   289→    }, 30_000);
   290→    return () => clearInterval(t);
   291→  }, []);
   292→
   293→  // ---- derived ----------------------------------------------------------- //
   294→  const ranked = useMemo(() => (snap ? computeRanked(snap, filter, now) : []), [snap, filter, now]);
   295→  const rankedIds = useMemo(() => ranked.map((l) => l.id), [ranked]);
   296→  const activePersona = personaDef(persona);
   297→
   298→  const openDossier = useCallback((lead: CommandLead) => {
   299→    setDossier(lead);
   300→    setDossierOpen(true);
   301→  }, []);
   302→
   303→  const openDossierById = useCallback(
   304→    (prospectId: string) => {
   305→      const lead = snap?.leads.find((l) => l.id === prospectId);
   306→      if (lead) openDossier(lead);
   307→    },
   308→    [snap, openDossier],
   309→  );
   310→
   311→  // ---- operator actions -------------------------------------------------- //
   312→  const doAction = useCallback(
   313→    async (prospectId: string, action: "called" | "deferred" | "snoozed" | "booked", note?: string) => {
   314→      const lead = snap?.leads.find((l) => l.id === prospectId);
   315→      try {
   316→        const res = await fetch("/api/actions", {
   317→          method: "POST",
   318→          headers: { "Content-Type": "application/json" },
   319→          body: JSON.stringify({ prospectId, action, note, hours: action === "snoozed" ? 24 : undefined }),
   320→        });
   321→        if (!res.ok) throw new Error(String(res.status));
   322→        pushToast({
   323→          kind: "info",
   324→          title:
   325→            action === "called" ? `Call logged — ${lead?.name ?? "prospect"}`
   326→            : action === "deferred" ? `Deferred — ${lead?.name ?? "prospect"}`
   327→            : action === "booked" ? `Booked — ${lead?.name ?? "prospect"}`
   328→            : `Snoozed 24h — ${lead?.name ?? "prospect"}`,
   329→          body: action === "deferred" ? "Returns to the queue on the next high-signal event." : undefined ?? "",
   330→          prospectId,
   331→        });
   332→        void refresh();
   333→      } catch {
   334→        pushToast({ kind: "info", title: "Action failed", body: "The enclave did not acknowledge — try again." });
   335→      }
   336→    },
   337→    [snap, pushToast, refresh],
   338→  );
   339→
   340→  const doTriage = useCallback(
   341→    async (draftId: string, decision: "approved" | "discarded") => {
   342→      try {
   343→        const res = await fetch("/api/replies", {
   344→          method: "PATCH",
   345→          headers: { "Content-Type": "application/json" },
   346→          body: JSON.stringify({ draftId, decision }),
   347→        });
   348→        if (!res.ok) throw new Error(String(res.status));
   349→        pushToast({
   350→          kind: decision === "approved" ? "info" : "info",
   351→          title: decision === "approved" ? "Draft approved & sent" : "Draft discarded",
   352→          body: decision === "approved" ? "Logged against the prospect timeline." : "The agent will not reuse this angle.",
   353→        });
   354→        void refresh();
   355→      } catch {
   356→        pushToast({ kind: "info", title: "Triage failed", body: "The enclave did not acknowledge — try again." });
   357→      }
   358→    },
   359→    [pushToast, refresh],
   360→  );
   361→
   362→  // ---- keyboard loop ----------------------------------------------------- //
   363→  useEffect(() => {
   364→    const onKey = (e: KeyboardEvent) => {
   365→      const target = e.target as HTMLElement | null;
   366→      if (target && (target.tagName === "INPUT" || target.tagName === "TEXTAREA" || target.isContentEditable)) return;
   367→
   368→      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === "k") {
   369→        e.preventDefault();
   370→        setPaletteOpen((o) => !o);
   371→        return;
   372→      }
   373→      if (e.key === "Escape") {
   374→        if (paletteOpen) setPaletteOpen(false);
   375→        else if (dossierOpen) setDossierOpen(false);
   376→        return;
   377→      }
   378→      if (paletteOpen || dossierOpen) return;
   379→
   380→      if (["1", "2", "3", "4", "5", "6"].includes(e.key)) {
   381→        setView(VIEW_KEYS[Number(e.key) - 1]);
   382→        return;
   383→      }
   384→      if (view !== "strike" || rankedIds.length === 0) return;
   385→
   386→      if (e.key === "j" || e.key === "k") {
   387→        e.preventDefault();
   388→        const idx = cursorId ? rankedIds.indexOf(cursorId) : -1;
   389→        const next = e.key === "j" ? Math.min(rankedIds.length - 1, idx + 1) : Math.max(0, idx <= 0 ? 0 : idx - 1);
   390→        setCursorId(rankedIds[next]);
   391→      } else if (e.key === "Enter") {
   392→        const lead = snap?.leads.find((l) => l.id === cursorId);
   393→        if (lead) openDossier(lead);
   394→      } else if (e.key === "c") {
   395→        const lead = snap?.leads.find((l) => l.id === cursorId);
   396→        const phone = lead?.contact?.phone;
   397→        if (phone) {
   398→          e.preventDefault();
   399→          window.location.href = `tel:${phone.replace(/\s/g, "")}`;
   400→        }
   401→      } else if (e.key === "x") {
   402→        if (cursorId) void doAction(cursorId, "deferred");
   403→      } else if (e.key === "s") {
   404→        if (cursorId) void doAction(cursorId, "snoozed");
   405→      } else if (e.key === "a") {
   406→        const first = snap?.triage.pending[0];
   407→        if (first?.draftId) void doTriage(first.draftId, "approved");
   408→      }
   409→    };
   410→    window.addEventListener("keydown", onKey);
   411→    return () => window.removeEventListener("keydown", onKey);
   412→  }, [view, rankedIds, cursorId, snap, paletteOpen, dossierOpen, doAction, doTriage, openDossier]);
   413→
   414→  // ---- render ------------------------------------------------------------ //
   415→  return (
   416→    <div className="command-root fixed inset-0 flex overflow-hidden bg-[#0B0C0B] font-sans text-[#F4F5F2] antialiased">
   417→      <CommandRail view={view} onView={setView} snap={snap} clock={clock || "--:--:--"} connected={connected} personaCode={activePersona.code} />
   418→
   419→      <main className="flex min-w-0 flex-1 flex-col">
   420→        <header className="flex h-14 shrink-0 items-center justify-between gap-4 border-b border-white/[0.08] px-6">
   421→          <div className="flex min-w-0 items-center gap-3">
   422→            <LiveDot level={connected ? "live" : "warn"} />
   423→            <p className="truncate font-mono text-[11px] uppercase tracking-[0.2em] text-[#8E9194]">{VIEW_TITLES[view]}</p>
   424→            <span className="hidden max-w-[430px] truncate whitespace-nowrap font-mono text-[10px] uppercase tracking-[0.14em] text-[#5F6460] xl:inline-block">
   425→              · {activePersona.title}: {activePersona.mandate}
   426→            </span>
   427→          </div>
   428→          <div className="flex shrink-0 items-center gap-3">
   429→            {/* persona switcher */}
   430→            <div className="hidden items-center gap-1 rounded-full border border-white/[0.10] bg-[#131513] p-1 md:flex" role="tablist" aria-label="Persona view">
   431→              {PERSONAS.map((p) => (
   432→                <button
   433→                  key={p.id}
   434→                  role="tab"
   435→                  aria-selected={persona === p.id}
   436→                  onClick={() => onPersona(p.id)}
   437→                  className={`h-7 rounded-full px-3 font-mono text-[9.5px] font-medium uppercase tracking-[0.1em] transition-all ${
   438→                    persona === p.id
   439→                      ? "bg-[#F4F5F2] text-[#0B0C0B]"
   440→                      : "text-[#8E9194] hover:text-[#F4F5F2]"
   441→                  }`}
   442→                >
   443→                  {p.code}
   444→                </button>
   445→              ))}
   446→            </div>
   447→            <HudMeta node="AST-00" hash="0a11" className="hidden xl:inline-flex" />
   448→            <button
   449→              onClick={() => setPaletteOpen(true)}
   450→              className="tactile-btn h-8 items-center gap-2 rounded-full border border-white/[0.12] bg-[#131513] px-3 font-mono text-[10.5px] uppercase tracking-[0.12em] text-[#8E9194] hover:text-[#F4F5F2] hover:border-white/25"
   451→            >
   452→              Jump <span className="rounded bg-[#1A1D1A] px-1.5 py-0.5 text-[9.5px]">⌘K</span>
   453→            </button>
   454→          </div>
   455→        </header>
   456→
   457→        {/* mobile view switcher — rail is hidden below md */}
   458→        <div
   459→          className="flex shrink-0 items-center gap-1.5 overflow-x-auto border-b border-white/[0.08] px-4 py-2 md:hidden"
   460→          role="tablist"
   461→          aria-label="Enclave views"
   462→        >
   463→          {VIEW_KEYS.map((k) => (
   464→            <button
   465→              key={k}
   466→              role="tab"
   467→              aria-selected={view === k}
   468→              onClick={() => setView(k)}
   469→              className={`h-7 shrink-0 rounded-full px-3 font-mono text-[9.5px] uppercase tracking-[0.1em] transition-all ${
   470→                view === k
   471→                  ? "bg-[#F4F5F2] text-[#0B0C0B]"
   472→                  : "border border-white/[0.12] bg-[#131513] text-[#8E9194]"
   473→              }`}
   474→            >
   475→              {k}
   476→            </button>
   477→          ))}
   478→        </div>
   479→
   480→        <div className="min-h-0 flex-1">
   481→          {!snap ? (
   482→            <BootScreen />
   483→          ) : (
   484→            <>
   485→              {view === "radar" && (
   486→                <RadarView
   487→                  snap={snap}
   488→                  now={now}
   489→                  freshEventIds={new Set([...fresh.entries()].filter(([, until]) => until > Date.now()).map(([id]) => id))}
   490→                  persona={persona}
   491→                  onOpenDossier={openDossier}
   492→                  onGoStrike={() => setView("strike")}
   493→                  onGoPipeline={() => setView("pipeline")}
   494→                  onGoPortal={() => setView("portal")}
   495→                />
   496→              )}
   497→              {view === "strike" && (
   498→                <StrikeView
   499→                  snap={snap}
   500→                  ranked={ranked}
   501→                  now={now}
   502→                  cursorId={cursorId}
   503→                  onCursor={setCursorId}
   504→                  onOpenDossier={openDossier}
   505→                  onAction={doAction}
   506→                  filter={filter}
   507→                  onFilter={(f) => {
   508→                    setFilter(f);
   509→                    setCursorId(null);
   510→                  }}
   511→                />
   512→              )}
   513→              {view === "theatre" && <TheatreView snap={snap} onOpenDossier={openDossier} />}
   514→              {view === "pipeline" && (
   515→                <PipelineView snap={snap} now={now} onOpenDossierId={openDossierById} onTriage={doTriage} />
   516→              )}
   517→              {view === "portal" && <PortalView snap={snap} />}
   518→              {view === "ops" && <OpsView snap={snap} />}
   519→            </>
   520→          )}
   521→        </div>
   522→      </main>
   523→
   524→      {/* Alert toasts — alert = deep onyx inverse, info = white tactile */}
   525→      <div className="pointer-events-none fixed right-5 top-5 z-[80] flex w-[380px] max-w-[calc(100vw-40px)] flex-col gap-2.5">
   526→        {toasts.map((t) => (
   527→          <button
   528→            key={t.id}
   529→            onClick={() => {
   530→              if (t.prospectId) openDossierById(t.prospectId);
   531→              setToasts((prev) => prev.filter((x) => x.id !== t.id));
   532→            }}
   533→            className={`cc-toast pointer-events-auto rounded-lg px-4 py-3.5 text-left transition-transform hover:scale-[1.01] ${
   534→              t.kind === "alert"
   535→                ? "ink shadow-[0_18px_50px_rgba(0,0,0,0.5)]"
   536→                : "neu border-white/[0.14] shadow-[0_18px_50px_rgba(0,0,0,0.5)]"
   537→            }`}
   538→          >
   539→            <div className="flex items-center gap-2">
   540→              {t.kind === "alert" && <span className="size-1.5 rounded-full bg-[#F4F5F2] cc-blink" />}
   541→              <p className={`font-mono text-[10.5px] font-medium uppercase tracking-[0.14em] ${t.kind === "alert" ? "text-white" : "text-[#8E9194]"}`}>
   542→                {t.title}
   543→              </p>
   544→            </div>
   545→            <p className={`mt-1.5 whitespace-pre-line text-[12px] leading-relaxed ${t.kind === "alert" ? "text-white/90" : "text-[#C9CCC7]"}`}>
   546→              {t.body}
   547→            </p>
   548→          </button>
   549→        ))}
   550→      </div>
   551→
   552→      <DossierDrawer lead={dossier} open={dossierOpen} onOpenChange={setDossierOpen} onAction={doAction} />
   553→
   554→      {/* Command palette */}
   555→      <Dialog open={paletteOpen} onOpenChange={setPaletteOpen}>
   556→        <DialogContent className="command-root top-[18%] max-w-[560px] translate-y-0 gap-0 rounded-lg border-white/[0.14] bg-[#131513] p-0 shadow-[0_40px_120px_-30px_rgba(0,0,0,0.75)] sm:top-[18%]">
   557→          <DialogTitle className="sr-only">Command palette</DialogTitle>
   558→          <Command className="bg-transparent">
   559→            <CommandInput placeholder="Jump to a view, a distributor, a draft…" className="border-b border-white/[0.10] font-mono text-[13px]" />
   560→            <CommandList className="cc-scroll max-h-[380px] p-2">
   561→              <CommandEmpty className="py-6 text-center font-mono text-[11px] uppercase tracking-[0.14em] text-[#5F6460]">
   562→                No match
   563→              </CommandEmpty>
   564→              <CommandGroup heading="Views" className="text-[#6E736E]">
   565→                {(Object.keys(VIEW_TITLES) as ViewKey[]).map((v) => (
   566→                  <CommandItem
   567→                    key={v}
   568→                    onSelect={() => {
   569→                      setView(v);
   570→                      setPaletteOpen(false);
   571→                    }}
   572→                    className="rounded-lg px-3 py-2.5 text-[13px] text-[#E9EBE7] data-[selected=true]:bg-[#222522] data-[selected=true]:text-[#F4F5F2]"
   573→                  >
   574→                    {VIEW_TITLES[v]}
   575→                  </CommandItem>
   576→                ))}
   577→              </CommandGroup>
   578→              {snap && snap.triage.pending.length > 0 && (
   579→                <CommandGroup heading="Pending drafts">
   580→                  {snap.triage.pending.map((t) => (
   581→                    <CommandItem
   582→                      key={t.replyId}
   583→                      onSelect={() => {
   584→                        setView("pipeline");
   585→                        setPaletteOpen(false);
   586→                      }}
   587→                      className="rounded-lg px-3 py-2.5 text-[13px] text-[#E9EBE7] data-[selected=true]:bg-[#222522] data-[selected=true]:text-[#F4F5F2]"
   588→                    >
   589→                      Review draft — {t.prospectName}
   590→                      <span className="ml-auto font-mono text-[10px] uppercase text-[#6E736E]">{t.intent.replace(/_/g, " ")}</span>
   591→                    </CommandItem>
   592→                  ))}
   593→                </CommandGroup>
   594→              )}
   595→              {snap && (
   596→                <CommandGroup heading="Distributors">
   597→                  {snap.leads.map((l) => (
   598→                    <CommandItem
   599→                      key={l.id}
   600→                      value={`${l.name} ${l.domain ?? ""} ${l.city ?? ""}`}
   601→                      onSelect={() => {
   602→                        openDossier(l);
   603→                        setPaletteOpen(false);
   604→                      }}
   605→                      className="rounded-lg px-3 py-2.5 text-[13px] text-[#E9EBE7] data-[selected=true]:bg-[#222522] data-[selected=true]:text-[#F4F5F2]"
   606→                    >
   607→                      {l.name}
   608→                      <span className="ml-auto font-mono text-[10px] tabular-nums text-[#6E736E]">
   609→                        {l.score?.qlvScore ?? "—"} · {l.city ?? "—"}
   610→                      </span>
   611→                    </CommandItem>
   612→                  ))}
   613→                </CommandGroup>
   614→              )}
   615→            </CommandList>
   616→          </Command>
   617→        </DialogContent>
   618→      </Dialog>
   619→    </div>
   620→  );
   621→}
