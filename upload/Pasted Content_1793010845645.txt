                          <div className="text-sm font-bold text-slate-100">{client.companyName}</div>
                          <div className="text-xs text-slate-400 mt-0.5">Payment Status: 
                            <span className={`ml-1 font-bold ${client.paymentStatus === 'PAID' ? 'text-emerald-400' : 'text-amber-400'}`}>{client.paymentStatus}</span>
                          </div>
                        </div>

                        <div className="text-right font-mono">
                          <div className="text-[10px] text-slate-500 uppercase">{revHorizon} 20% Cut</div>
                          <div className="text-lg font-bold text-sky-400">
                            ${(revHorizon === 'monthly' ? client.monthlyRevShare : revHorizon === 'annual' ? client.annualRevShare : client.lifetimeRevShare).toLocaleString()}
                          </div>
                        </div>
                      </div>
                    ))}
                  </div>
                </div>
              )}

              {/* MODAL 4: AUDIT GUARANTEE STATUS & SALES LOG */}
              {activeWidgetModal === 'guarantee' && (
                <div className="bg-slate-900 border border-emerald-500/40 rounded-3xl p-8 space-y-6 animate-fadeIn">
                  <div className="flex justify-between items-center border-b border-slate-800 pb-4">
                    <div>
                      <span className="text-xs font-mono text-emerald-400 uppercase tracking-widest">[ AUDIT GUARANTEE VAULT ]</span>
                      <h3 className="text-2xl font-light text-slate-100 mt-1">Audit Sales & Conversion History</h3>
                    </div>
                    <button onClick={() => setActiveWidgetModal(null)} className="text-slate-500 hover:text-white font-mono text-xs">[Close Panel]</button>
                  </div>

                  <div className="space-y-3">
                    {auditRecords.map((audit) => (
                      <div key={audit.id} className="bg-slate-950 border border-slate-800 p-4 rounded-xl space-y-3">
                        <div className="flex justify-between items-start">
                          <div>
                            <div className="text-sm font-bold text-slate-100">{audit.companyName}</div>
                            <div className="text-xs text-slate-400">Date Sold: {audit.dateSold} · Fee: ${audit.auditFee}</div>
                          </div>
                          <span className={`text-[10px] px-2.5 py-1 rounded-full border ${audit.convertedToClient ? 'bg-emerald-950 text-emerald-400 border-emerald-500/30' : 'bg-slate-900 text-slate-400 border-slate-800'}`}>
                            {audit.convertedToClient ? 'Converted ($50k Executed)' : 'Pending Conversion'}
                          </span>
                        </div>

                        <div className="text-xs font-mono bg-slate-900 p-3 rounded-lg text-slate-300 space-y-1">
                          <div className="flex justify-between">
                            <span className="text-slate-500">Briefing Call Host:</span>
                            <span className="text-slate-200">{audit.callHost}</span>
                          </div>
                          <div className="flex justify-between">
                            <span className="text-slate-500">Attended Call:</span>
                            <span className={audit.attendedCall ? 'text-emerald-400 font-bold' : 'text-rose-400 font-bold'}>
                              {audit.attendedCall ? 'Yes (Attended)' : 'No (Missed Session)'}
                            </span>
                          </div>
                          <div className="flex justify-between">
                            <span className="text-slate-500">Identified Profit Leakage:</span>
                            <span className="text-emerald-400 font-bold">${audit.identifiedLeakage.toLocaleString()}</span>
                          </div>
                        </div>

                        <div className="text-xs text-slate-400 font-sans italic border-t border-slate-900 pt-2">
                          "{audit.callNotes}"
                        </div>
                      </div>
                    ))}
                  </div>
                </div>
              )}

              {/* DEEP CLIENT ENCLAVE MODAL VIEW */}
              {selectedClientForModal && (
                <div className="fixed inset-0 z-50 bg-slate-950/80 backdrop-blur-md flex items-center justify-center p-6">
                  <div className="bg-slate-900 border border-emerald-500/40 rounded-3xl p-8 max-w-2xl w-full space-y-6 shadow-2xl font-mono">
                    <div className="flex justify-between items-start border-b border-slate-800 pb-4">
                      <div>
                        <span className="text-xs text-emerald-400 uppercase font-bold">[ CLIENT ENCLAVE VIEW ]</span>
                        <h3 className="text-2xl font-light text-slate-100 mt-1">{selectedClientForModal.companyName}</h3>
                        <div className="text-xs text-slate-400 mt-0.5">{selectedClientForModal.erpSystem} · AST Isolation Active</div>
                      </div>
                      <button onClick={() => setSelectedClientForModal(null)} className="text-slate-500 hover:text-white text-xs">[Close View]</button>
                    </div>

                    <div className="grid grid-cols-2 gap-4 text-xs">
                      <div className="bg-slate-950 p-4 rounded-xl border border-slate-800">
                        <div className="text-slate-500">Total Defended Margin</div>
                        <div className="text-2xl font-bold text-emerald-400 mt-1">${selectedClientForModal.totalDefended.toLocaleString()}</div>
                      </div>
                      <div className="bg-slate-950 p-4 rounded-xl border border-slate-800">
                        <div className="text-slate-500">Monthly Rev-Share (20%)</div>
                        <div className="text-2xl font-bold text-sky-400 mt-1">${selectedClientForModal.monthlyRevShare.toLocaleString()}</div>
                      </div>
                    </div>

                    <div className="bg-slate-950 p-4 rounded-xl border border-slate-800 text-xs space-y-2">
                      <div className="text-slate-400 font-bold uppercase">Primary Margin Leakage Vector:</div>
                      <div className="text-slate-200">{selectedClientForModal.primaryLeakageVector}</div>
                    </div>

                    <button onClick={() => setSelectedClientForModal(null)} className="w-full bg-emerald-500 hover:bg-emerald-400 text-slate-950 font-bold py-3 rounded-xl text-xs transition">
                      RETURN TO COMMAND TOWER
                    </button>
                  </div>
                </div>
              )}

            </div>
          )}

        </main>
      </div>

    </div>
  );
}
