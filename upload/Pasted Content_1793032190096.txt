// Edge-safe HMAC sessions (operator + tenant scopes). Phases 1 & 9.
const enc = new TextEncoder();
const b64u = {
  enc: (buf: ArrayBuffer) => btoa(String.fromCharCode(...new Uint8Array(buf))).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, ''),
  dec: (s: string) => Uint8Array.from(atob(s.replace(/-/g, '+').replace(/_/g, '/')), (c) => c.charCodeAt(0)),
};
export const SESSION_COOKIE = 'syntropic_session';
export const TENANT_COOKIE = 'syntropic_tenant';
const SESSION_TTL_MS = 7 * 24 * 3600e3;
const TENANT_SESSION_TTL_MS = 14 * 24 * 3600e3;
export const TENANT_LINK_TTL_MS = 7 * 24 * 3600e3;

export function operatorEmails(): string[] {
  return (process.env.OPERATOR_EMAILS ?? '').split(',').map((s) => s.trim().toLowerCase()).filter(Boolean);
}
export const isOperator = (email: string) => operatorEmails().includes(email.toLowerCase());

async function hmacKey() {
  const secret = process.env.AUTH_SECRET ?? 'dev-insecure-secret-change-me';
  return crypto.subtle.importKey('raw', enc.encode(secret), { name: 'HMAC', hash: 'SHA-256' }, false, ['sign', 'verify']);
}
export async function signSession(email: string): Promise<string> {
  const payload = { scope: 'operator', email: email.toLowerCase(), exp: Date.now() + SESSION_TTL_MS, jti: crypto.randomUUID() };
  const body = b64u.enc(enc.encode(JSON.stringify(payload)).buffer as ArrayBuffer);
  return `${body}.${b64u.enc(await crypto.subtle.sign('HMAC', await hmacKey(), enc.encode(body)))}`;
}
export async function verifySession(token: string | undefined | null) {
  if (!token || !token.includes('.')) return null;
  const [body, sig] = token.split('.');
  try {
    if (!(await crypto.subtle.verify('HMAC', await hmacKey(), b64u.dec(sig), enc.encode(body)))) return null;
    const p = JSON.parse(new TextDecoder().decode(b64u.dec(body)));
    if (p.scope !== 'operator' || typeof p.exp !== 'number' || p.exp < Date.now()) return null;
    if (!isOperator(p.email)) return null;
    return { email: p.email };
  } catch { return null; }
}
export async function signTenantSession(tenantId: string, email: string): Promise<string> {
  const payload = { scope: 'tenant', tenantId, email: email.toLowerCase(), exp: Date.now() + TENANT_SESSION_TTL_MS, jti: crypto.randomUUID() };
  const body = b64u.enc(enc.encode(JSON.stringify(payload)).buffer as ArrayBuffer);
  return `${body}.${b64u.enc(await crypto.subtle.sign('HMAC', await hmacKey(), enc.encode(body)))}`;
}
export async function verifyTenantSession(token: string | undefined | null) {
  if (!token || !token.includes('.')) return null;
  const [body, sig] = token.split('.');
  try {
    if (!(await crypto.subtle.verify('HMAC', await hmacKey(), b64u.dec(sig), enc.encode(body)))) return null;
    const p = JSON.parse(new TextDecoder().decode(b64u.dec(body)));
    if (p.scope !== 'tenant' || typeof p.exp !== 'number' || p.exp < Date.now()) return null;
    return { tenantId: p.tenantId as string, email: p.email as string };
  } catch { return null; }
}
export async function hashToken(raw: string): Promise<string> {
  return [...new Uint8Array(await crypto.subtle.digest('SHA-256', enc.encode(raw)))]
    .map((b) => b.toString(16).padStart(2, '0')).join('');
}
export function randomToken(bytes = 32): string {
  const b = new Uint8Array(bytes); crypto.getRandomValues(b);
  return [...b].map((x) => x.toString(16).padStart(2, '0')).join('');
}

// lib/geo.ts — geo resolution: edge headers → cached provider → env fallback. Phase 11.8.
export type Geo = { city: string | null; country: string | null };
export type GeoSource = 'edge' | 'provider' | 'fallback' | 'private' | 'none';

const CACHE = new Map<string, { geo: Geo; at: number }>();
const CACHE_TTL_MS = 24 * 3600e3;
const CACHE_MAX = 5000;
const PRIVATE_RE = /^(10\.|127\.|192\.168\.|172\.(1[6-9]|2\d|3[01])\.|169\.254\.|::1|f[cd][0-9a-f]{2}:)/i;

/** "Sanitized IP": port-stripped; IPv4 last octet zeroed / IPv6 → /64 unless IP_PRIVACY_MODE=full. 'off' stores nothing. */
export function sanitizeIp(raw: string | null): string | null {
  if (!raw) return null;
  let ip = raw.trim();
  const v4port = ip.match(/^(\d{1,3}(?:\.\d{1,3}){3}):\d+$/);
  if (v4port) ip = v4port[1];
  const mode = process.env.IP_PRIVACY_MODE ?? 'truncated';
  if (mode === 'off') return null;
  if (mode === 'full') return ip;
  if (ip.includes(':')) return ip.split(':').slice(0, 4).join(':') + '::';
  const o = ip.split('.');
  return o.length === 4 ? `${o[0]}.${o[1]}.${o[2]}.0` : ip;
}

function envFallback(): Geo {
  return { city: process.env.GEO_FALLBACK_CITY || null, country: process.env.GEO_FALLBACK_COUNTRY || null };
}

/** PRIMARY: edge-injected headers (Vercel / Cloudflare / Nginx geoip). Authoritative when present. */
export function geoFromHeaders(h: Headers): Geo | null {
  const rawCity = h.get('x-vercel-ip-city') ?? h.get('cf-ipcity') ?? h.get('x-geo-city');
  const rawCountry = h.get('x-vercel-ip-country') ?? h.get('cf-ipcountry') ?? h.get('x-geo-country');
  if (!rawCity && !rawCountry) return null;
  let city: string | null = null;
  if (rawCity) { try { city = decodeURIComponent(rawCity); } catch { city = rawCity; } }
  return { city, country: rawCountry ? rawCountry.toUpperCase() : null }; // ISO-3166 alpha-2, never 'INT'
}

/** FALLBACK: cached external lookup for self-hosted deployments (no edge headers).
    Miss returns env fallback NOW, refreshes cache in background. GEO_PROVIDER=none disables egress. */
export function getGeoByIp(fullIp: string | null): { geo: Geo; source: GeoSource } {
  if (!fullIp || PRIVATE_RE.test(fullIp)) return { geo: { city: null, country: null }, source: 'private' };
  const hit = CACHE.get(fullIp);
  if (hit && Date.now() - hit.at < CACHE_TTL_MS) return { geo: hit.geo, source: 'provider' };
  if ((process.env.GEO_PROVIDER ?? 'ip-api') === 'none') return { geo: envFallback(), source: 'fallback' };

  const ctrl = new AbortController();
  const kill = setTimeout(() => ctrl.abort(), 2500);
  fetch(`http://ip-api.com/json/${encodeURIComponent(fullIp)}?fields=city,countryCode,status`, { signal: ctrl.signal })
    .then((r) => (r.ok ? r.json() : null))
    .then((d) => {
      if (!d || d.status !== 'success') return;
      if (CACHE.size >= CACHE_MAX) CACHE.delete(CACHE.keys().next().value as string);
      CACHE.set(fullIp, { geo: { city: d.city || null, country: d.countryCode || null }, at: Date.now() });
    })
    .catch(() => {})
    .finally(() => clearTimeout(kill));

  return hit ? { geo: hit.geo, source: 'provider' } : { geo: envFallback(), source: 'fallback' };
}

/** Single entry point for every route: edge headers win, provider fallback second. */
export function resolveGeo(h: Headers, ip: string | null): { geo: Geo; source: GeoSource } {
  const edge = geoFromHeaders(h);
  if (edge) return { geo: edge, source: 'edge' };
  return getGeoByIp(ip);
}

import { createHash } from 'node:crypto';
import { prisma } from './prisma';
import { sanitizeIp } from './geo';
import { emit } from './bus';

export const TRAFFIC_LAYERS = ['LANDING', 'PORTAL', 'ENGINE'] as const;
export type TrafficLayer = (typeof TRAFFIC_LAYERS)[number];
export type TrafficEventType =
  | 'PAGE_VIEW' | 'SLIDER_CHANGE' | 'CTA_CLICK' | 'SCROLL_DEPTH'
  | 'INTAKE_SUBMITTED' | 'DEPOSIT_CLICKED' | 'PAYMENT_SUCCESS'
  | 'CALL_BOOKED' | 'PORTAL_LOGIN' | 'FILE_UPLOAD' | 'AUDIT_SUBMITTED'
  | 'BRIEFING_ATTENDED' | 'CONVERSION_RECORDED';

function daySalt(): string {
  const days = Math.max(0, Number(process.env.IP_HASH_SALT_ROTATE_DAYS ?? 30) || 0);
  const epoch = Math.floor(Date.now() / (days * 86400e3 || 86400e3));
  return `${process.env.IP_HASH_SALT_BASE ?? 'dev-salt'}:${epoch}`;
}
export function hashIp(ip: string | null): string | null {
  if (!ip) return null;
  return createHash('sha256').update(`${ip}|${daySalt()}`).digest('hex');
}
export const utcDay = (d = new Date()) => d.toISOString().slice(0, 10);

/** Fire-and-forget capture. Never throws into the caller's request path. */
export async function recordTraffic(input: {
  sessionId: string; layer: TrafficLayer; eventType: TrafficEventType;
  ip: string | null; tenantId?: string | null; metadata?: Record<string, unknown>;
  geo?: { city: string | null; country: string | null };
}): Promise<void> {
  try {
    const day = utcDay();
    await prisma.$transaction([
      prisma.trafficEvent.create({
        data: {
          sessionId: input.sessionId,
          tenantId: input.tenantId ?? null,
          layer: input.layer,
          eventType: input.eventType,
          ipHash: hashIp(input.ip),
          ipAddress: sanitizeIp(input.ip),
          city: input.geo?.city ?? null,
          country: input.geo?.country ?? null,
          metadata: input.metadata ? JSON.stringify(input.metadata) : null,
        },
      }),
      prisma.liveTelemetrySummary.upsert({
        where: { date: day },
        create: { date: day, totalVisitors: 1 },
        update: { totalVisitors: { increment: 1 } },
      }),
    ]);
    if (input.eventType === 'INTAKE_SUBMITTED') await bump(day, 'auditsInitiated');
    if (input.eventType === 'PORTAL_LOGIN')       await bump(day, 'portalLogins');
    if (input.eventType === 'FILE_UPLOAD')        await bump(day, 'quotesIngested');
    if (input.layer === 'LANDING' || input.layer === 'PORTAL') {
      emit({ type: 'traffic', city: input.geo?.city ?? null, country: input.geo?.country ?? null, eventType: input.eventType });
    }
  } catch (e) { console.error('[traffic] capture failed (non-fatal):', e); }
}

async function bump(date: string, field: 'auditsInitiated' | 'portalLogins' | 'quotesIngested') {
  try { await prisma.liveTelemetrySummary.update({ where: { date }, data: { [field]: { increment: 1 } } as never }); }
  catch { /* row vanished — recalc heals */ }
}

/** Unique sessions & visitors for a day — recomputed from the event log. */
export async function recomputeDay(date = utcDay()) {
  const events = await prisma.trafficEvent.findMany({
    where: { createdAt: { gte: new Date(`${date}T00:00:00.000Z`), lt: new Date(`${date}T23:59:59.999Z`) } },
    select: { sessionId: true, ipHash: true },
  });
  const summary = await prisma.liveTelemetrySummary.findUnique({ where: { date } });
  await prisma.liveTelemetrySummary.upsert({
    where: { date },
    create: { date, totalVisitors: events.length, uniqueSessions: new Set(events.map((e) => e.sessionId)).size },
    update: { totalVisitors: events.length, uniqueSessions: new Set(events.map((e) => e.sessionId)).size,
              auditsInitiated: summary?.auditsInitiated ?? 0, portalLogins: summary?.portalLogins ?? 0,
              quotesIngested: summary?.quotesIngested ?? 0 },
  });
}

import { prisma } from './prisma';
export async function assignFounder() {
  const members = await prisma.teamMember.findMany({ where: { isOnline: true } });
  const pool = members.length ? members : await prisma.teamMember.findMany();
  if (!pool.length) throw new Error('No TeamMembers seeded — run `npx prisma db seed`');
  const active = await prisma.tenant.groupBy({
    by: ['assignedToId'], _count: { _all: true },
    where: { assignedToId: { in: pool.map((m) => m.id) }, status: { not: 'CONVERTED' } },
  });
  const load = new Map(active.map((a) => [a.assignedToId, a._count._all]));
  return pool.sort((a, b) => (load.get(a.id) ?? 0) - (load.get(b.id) ?? 0) || a.auditsClosed - b.auditsClosed)[0];
}

const buckets = new Map<string, { tokens: number; last: number }>();
export function rateLimit(key: string, limit: number, windowMs: number): boolean {
  const now = Date.now();
  const b = buckets.get(key) ?? { tokens: limit, last: now };
  b.tokens = Math.min(limit, b.tokens + ((now - b.last) / windowMs) * limit);
  b.last = now;
  if (b.tokens < 1) { buckets.set(key, b); return false; }
  b.tokens -= 1; buckets.set(key, b); return true;
}

export interface ParsedLine { lineNumber: number; rawText: string; confidence: number }
export const HITL_CONFIDENCE_FLOOR = 0.95;
export async function parseQuoteFile(fileName: string): Promise<ParsedLine[]> {
  const count = 3 + (fileName.length % 5);
  const lines: ParsedLine[] = [];
  for (let i = 0; i < count; i++) {

    lines.push({
      lineNumber: i + 1,
      rawText: `${fileName} — line ${i + 1} · sample part shorthand (Docling demo)`,

  return lines;

  if (!form) return NextResponse.json({ error: 'expected_multipart' }, { status: 400 });

  if (!tenant) return NextResponse.json({ error: 'tenant_not_found' }, { status: 401 });
  if (tenant.paymentStatus !== 'PAID') {
    return NextResponse.json({ error: 'ingest_locked', detail: 'File ingest requires an active $3,500 Diagnostic Audit deposit.' }, { status: 402 });
  }

  const files = form.getAll('files').filter((f): f is File => f instanceof File);
  const namedOnly = String(form.get('namesOnly') ?? '');

    return NextResponse.json({ error: 'no_files', detail: 'File contents missing — re-add the files in your enclave and re-submit.' }, { status: 422 });
  }

  let queued = 0;
  for (const f of files) {
    if (!/\.(pdf|xlsx|csv)$/i.test(f.name) || f.size > 50 * 1048576) continue;

    await writeFile(path.join(dir, `${Date.now()}_${safe}`), Buffer.from(await f.arrayBuffer()));

      hitlRows.push({ fileName: f.name, lineNumber: line.lineNumber, rawText: line.rawText, confidence: line.confidence });
    }

    }
  }

    metadata: { queued, hitl: hitlRows.length } });
  return NextResponse.json({ ok: true, tenantId: tenant.id, filesQueued: queued, hitlCount: hitlRows.length });
}

  });

  });

  });

        where: { tenantId: { in: tenantIds }, city: { not: null } },
        orderBy: { createdAt: 'desc' },

      })
    : [];
  const tenantCity = new Map<string, string>();
  for (const e of latestPerTenant) {
    if (e.tenantId && !tenantCity.has(e.tenantId)) tenantCity.set(e.tenantId, `${e.city}|${e.country}`);
  }
  const enclaveByCity = new Map<string, { city: string; country: string; tenants: number; reviews: number }>();
  for (const t of activeTenants) {
    const key = tenantCity.get(t.id);
    if (!key) continue;
    const [city, country] = key.split('|');
    const cur = enclaveByCity.get(key) ?? { city, country, tenants: 0, reviews: 0 };

    cur.reviews += t.hitlItems.length;

  }

import { PrismaClient } from '@prisma/client';
import { createHash, randomBytes } from 'node:crypto';
const prisma = new PrismaClient();
const sha = (s: string) => createHash('sha256').update(s).digest('hex');
const email = process.argv[2]?.toLowerCase();
if (!email) { console.error('usage: npx tsx scripts/issue-tenant-link.ts <tenant-email>'); process.exit(1); }
const tenant = await prisma.tenant.findUnique({ where: { contactEmail: email } });
if (!tenant) { console.error('tenant not found — run an intake first'); process.exit(1); }
const raw = randomBytes(32).toString('hex');
await prisma.magicLinkToken.create({
  data: { email, tokenHash: sha(raw), scope: 'tenant', tenantId: tenant.id, expiresAt: new Date(Date.now() + 7 * 24 * 3600e3) },
});
console.log(`${process.env.PUBLIC_BASE_URL ?? 'http://localhost:3000'}/api/auth/tenant-callback?token=${raw}`);
await prisma.$disconnect();
